#+TITLE: Apple Reminders — org-reminders Setup
#+STARTUP: overview

Loaded from [[file:config.org][config.org]] on macOS only (=system-type 'darwin=).
Not loaded in Docker/Orbstack variants (=system-type 'gnu/linux=).

Package: [[https://github.com/ginqi7/org-reminders][ginqi7/org-reminders]] — bidirectional sync between Apple Reminders and Org.
Replaces the earlier =apple-reminders.org= custom implementation.

* Prerequisites

- *org-reminders-cli*: =brew install ginqi7/formulae/org-reminders-cli= — installed automatically
- *websocket-bridge* + *websocket*: Emacs packages — installed automatically
- *Reminders access*: must be granted manually once (see below)

* Grant Reminders Access (once)

macOS grants Reminders permission per process. Since the CLI is launched by Emacs,
the permission must be granted to the *Emacs process* — not Terminal.

Run this command in *Eshell inside Emacs* (=M-x eshell=):

#+begin_src shell :tangle no
osascript -e 'tell application "Reminders" to show (first list whose name is "blabla")'
#+end_src

macOS shows a permission dialog → click *OK*. Then once:

#+begin_src emacs-lisp :tangle no
(org-reminders-restart-auto-sync)
#+end_src

* Installation & Configuration

#+begin_src emacs-lisp :tangle yes
;;; Apple Reminders via org-reminders — macOS only
;;; Dependencies: org-reminders-cli (brew) + websocket-bridge (GitHub) + websocket (MELPA)

;; Register agenda file first — stays active even if packages below fail to load
(add-to-list 'org-agenda-files (expand-file-name "~/org/reminders.org"))

;; Install org-reminders-cli automatically if missing
(unless (executable-find "org-reminders")
  (my/brew-install-and-log "formula" "org-reminders-cli"
                           "install" "ginqi7/formulae/org-reminders-cli"))

;; quelpa: installs GitHub packages and activates them in the current session
;; (package-vc-install does not activate immediately → compile errors with dependencies)
(use-package quelpa
  :ensure t
  :custom
  (quelpa-update-melpa-p nil)
  (quelpa-self-upgrade-p nil))

;; websocket (MELPA) — dependency of websocket-bridge
(use-package websocket
  :ensure t)

;; Install websocket-bridge first (dependency of org-reminders)
(unless (package-installed-p 'websocket-bridge)
  (quelpa '(websocket-bridge :fetcher github :repo "ginqi7/websocket-bridge")))

;; Install org-reminders
(unless (package-installed-p 'org-reminders)
  (quelpa '(org-reminders :fetcher github :repo "ginqi7/org-reminders")))

(use-package org-reminders
  :ensure nil
  :custom
  (org-reminders-sync-file (expand-file-name "~/org/reminders.org"))
  ;; "all" | "incomplete" | "complete"
  (org-reminders-display-options "all")
  ;; Number of saves before a sync is triggered (1 = every save)
  (org-reminders-sync-frequency 1))

;; First start: run sync-all if reminders.org has no data yet, then start auto-sync
;; via process sentinel once sync-all completes.
;; Subsequent starts: go straight to auto-sync.
(when (executable-find "org-reminders")
  (run-with-idle-timer 3 nil
                       (lambda ()
                         (org-reminders-check-sync-file)
                         (if (with-temp-buffer
                               (insert-file-contents org-reminders-sync-file)
                               (not (search-forward "EXTERNAL-ID" nil t)))
                             ;; First run: sync-all, start auto-sync after completion
                             (let ((proc (org-reminders-sync-all)))
                               (when (processp proc)
                                 (set-process-sentinel
                                  proc
                                  (lambda (p _m)
                                    (when (zerop (process-exit-status p))
                                      (org-reminders-start-auto-sync))))))
                           ;; Subsequent starts: start auto-sync directly
                           (org-reminders-start-auto-sync)))))

;; Fix: package skips the org-priority call when Apple priority is 0 (no priority)
;; → existing priority in the org entry is never cleared when removed in Apple
(with-eval-after-load 'org-reminders
  (advice-add 'org-reminders--item-update-detail :after
              (lambda (obj)
                (when (= 0 (org-reminders-item-priority obj))
                  (org-priority 'remove)))))
#+end_src

* Emacs→Apple Sync (disabled — reference)

This block fixes the issue where changes made in Emacs do not arrive in Apple Reminders.
Disabled while testing the default package behavior.

Problem: =org-reminders-with-sync-file= calls =save-buffer= internally — without the
guard flag the =after-save-hook= would fire recursively (loop).

#+begin_src emacs-lisp :tangle no
;; Flag: t while package is applying Apple→Org updates → prevents sync loop
(defvar my/org-reminders--from-apple nil)

(with-eval-after-load 'org-reminders
  (advice-add 'org-reminders-message-handle :around
              (lambda (orig &rest args)
                (let ((my/org-reminders--from-apple t))
                  (apply orig args))))
  (advice-add 'org-reminders-check-sync-file :around
              (lambda (orig &rest args)
                (let ((my/org-reminders--from-apple t))
                  (apply orig args)))))

;; Update LAST-MODIFIED when user changes TODO state → syncOnce detects the change
(defun my/org-reminders-mark-entry-modified ()
  (when (and (not my/org-reminders--from-apple)
             (buffer-file-name)
             (string= (buffer-file-name)
                      (expand-file-name "~/org/reminders.org"))
             (org-entry-get nil "EXTERNAL-ID"))
    (org-entry-put nil "LAST-MODIFIED"
                   (format-time-string "%Y-%m-%d %H:%M:%S"))))

(add-hook 'org-after-todo-state-change-hook #'my/org-reminders-mark-entry-modified)

;; Send sync-once after user save — skipped for package-internal saves (from-apple = t)
(defvar my/org-reminders--sync-timer nil)

(defun my/org-reminders-on-save ()
  (when (and (not my/org-reminders--from-apple)
             (buffer-file-name)
             (string= (buffer-file-name)
                      (expand-file-name "~/org/reminders.org")))
    (when my/org-reminders--sync-timer
      (cancel-timer my/org-reminders--sync-timer))
    (setq my/org-reminders--sync-timer
          (run-with-timer 1 nil
                          (lambda ()
                            (condition-case err
                                (org-reminders-send-message "sync-once")
                              (error
                               (message "org-reminders sync-once: %s"
                                        (error-message-string err)))))))))

(add-hook 'after-save-hook #'my/org-reminders-on-save)

;; Auto-save after 3s idle → no manual C-x C-s needed
(defun my/org-reminders-enable-autosave ()
  (when (and (buffer-file-name)
             (string= (buffer-file-name)
                      (expand-file-name "~/org/reminders.org")))
    (setq-local auto-save-visited-interval 3)
    (auto-save-visited-mode 1)))

(add-hook 'find-file-hook #'my/org-reminders-enable-autosave)
#+end_src
