#+TITLE: Apple Reminders — org-reminders Setup
#+STARTUP: overview

Geladen von [[file:config.org][config.org]] nur auf macOS (=system-type 'darwin=).
Nicht geladen in Docker/Orbstack-Varianten (=system-type 'gnu/linux=).

Package: [[https://github.com/ginqi7/org-reminders][ginqi7/org-reminders]] — bidirektionaler Sync zwischen Apple Reminders und Org.
Ersetzt die frühere =apple-reminders.org= Eigenimplementierung.

* Voraussetzungen

- *org-reminders-cli*: =brew install ginqi7/formulae/org-reminders-cli= — wird automatisch installiert
- *websocket-bridge* + *websocket*: Emacs-Pakete — werden automatisch installiert
- *Reminders-Zugriff*: macOS fragt beim ersten Sync nach Berechtigung

* Installation & Konfiguration

#+begin_src emacs-lisp :tangle yes
;;; Apple Reminders via org-reminders — nur macOS
;;; Abhängigkeiten: org-reminders-cli (brew) + websocket-bridge (GitHub) + websocket (MELPA)

;; Agenda-Eintrag zuerst — damit er auch bei Ladefehlern weiter unten aktiv bleibt
(add-to-list 'org-agenda-files (expand-file-name "~/org/reminders.org"))

;; org-reminders-cli automatisch im Hintergrund installieren falls fehlend
(unless (executable-find "org-reminders")
  (my/brew-install-and-log "formula" "org-reminders-cli"
                           "install" "ginqi7/formulae/org-reminders-cli"))

;; quelpa: installiert GitHub-Packages und aktiviert sie sofort in der Session
;; (package-vc-install aktiviert nicht sofort → Kompilierungsfehler bei Abhängigkeiten)
(use-package quelpa
  :ensure t
  :custom
  (quelpa-update-melpa-p nil)
  (quelpa-self-upgrade-p nil))

;; websocket (MELPA) — Abhängigkeit von websocket-bridge
(use-package websocket
  :ensure t)

;; websocket-bridge zuerst installieren (Abhängigkeit von org-reminders)
(unless (package-installed-p 'websocket-bridge)
  (quelpa '(websocket-bridge :fetcher github :repo "ginqi7/websocket-bridge")))

;; org-reminders installieren
(unless (package-installed-p 'org-reminders)
  (quelpa '(org-reminders :fetcher github :repo "ginqi7/org-reminders")))

(use-package org-reminders
  :ensure nil
  :custom
  (org-reminders-sync-file (expand-file-name "~/org/reminders.org")))

;; Auto-Sync starten sobald CLI verfügbar (idle damit Startup nicht blockiert)
(when (executable-find "org-reminders")
  (run-with-idle-timer 3 nil
                       (lambda ()
                         (org-reminders-check-sync-file)
                         (org-reminders-start-auto-sync))))

;; before-save: LAST-MODIFIED aller Einträge auf jetzt setzen damit
;; syncOnce() erkennt dass org neuer ist als Apple
(defun my/org-reminders-update-last-modified ()
  (when (and (buffer-file-name)
             (string= (buffer-file-name)
                      (expand-file-name "~/org/reminders.org")))
    (let ((now (format-time-string "%Y-%m-%d %H:%M:%S")))
      (org-map-entries
       (lambda ()
         (when (org-entry-get nil "EXTERNAL-ID")
           (org-entry-put nil "LAST-MODIFIED" now)))
       nil nil))))

(add-hook 'before-save-hook #'my/org-reminders-update-last-modified)

;; after-save: CLI sync --type once als Subprocess (bidirektional, kein WebSocket nötig)
(defun my/org-reminders-on-save ()
  (when (and (buffer-file-name)
             (string= (buffer-file-name)
                      (expand-file-name "~/org/reminders.org")))
    (start-process "org-reminders-sync" nil
                   (executable-find "org-reminders")
                   "sync"
                   (expand-file-name "~/org/reminders.org")
                   "--type" "once")))

(add-hook 'after-save-hook #'my/org-reminders-on-save)

;; Auto-Save nach 3s Idle → triggert after-save-hook → kein manuelles C-x C-s nötig
(defun my/org-reminders-enable-autosave ()
  (when (and (buffer-file-name)
             (string= (buffer-file-name)
                      (expand-file-name "~/org/reminders.org")))
    (setq-local auto-save-visited-interval 3)
    (auto-save-visited-mode 1)))

(add-hook 'find-file-hook #'my/org-reminders-enable-autosave)
#+end_src
