#+TITLE: Apple Reminders — org-reminders Setup
#+STARTUP: overview

Geladen von [[file:config.org][config.org]] nur auf macOS (=system-type 'darwin=).
Nicht geladen in Docker/Orbstack-Varianten (=system-type 'gnu/linux=).

Package: [[https://github.com/ginqi7/org-reminders][ginqi7/org-reminders]] — bidirektionaler Sync zwischen Apple Reminders und Org.
Ersetzt die frühere =apple-reminders.org= Eigenimplementierung.

* Voraussetzungen

- *org-reminders-cli*: =brew install ginqi7/formulae/org-reminders-cli= — wird automatisch installiert
- *websocket-bridge* + *websocket*: Emacs-Pakete — werden automatisch installiert
- *Reminders-Zugriff*: muss einmalig manuell erteilt werden (siehe unten)

* Reminders-Zugriff erteilen (einmalig)

macOS vergibt Reminders-Berechtigung pro Prozess. Da der CLI von Emacs gestartet
wird, muss die Berechtigung dem *Emacs-Prozess* erteilt werden — nicht Terminal.

Diesen Befehl in *Eshell innerhalb von Emacs* ausführen (=M-x eshell=):

#+begin_src shell
osascript -e 'tell application "Reminders" to show (first list whose name is "blabla")'
#+end_src

macOS zeigt einen Berechtigungsdialog → *OK* klicken. Danach einmalig:

#+begin_src emacs-lisp :tangle no
(org-reminders-restart-auto-sync)
#+end_src

* Installation & Konfiguration

#+begin_src emacs-lisp :tangle yes
;;; Apple Reminders via org-reminders — nur macOS
;;; Abhängigkeiten: org-reminders-cli (brew) + websocket-bridge (GitHub) + websocket (MELPA)

;; Agenda-Eintrag zuerst — damit er auch bei Ladefehlern weiter unten aktiv bleibt
(add-to-list 'org-agenda-files (expand-file-name "~/org/reminders.org"))

;; org-reminders-cli automatisch im Hintergrund installieren falls fehlend
(unless (executable-find "org-reminders")
  (my/brew-install-and-log "formula" "org-reminders-cli"
                           "install" "ginqi7/formulae/org-reminders-cli"))

;; quelpa: installiert GitHub-Packages und aktiviert sie sofort in der Session
;; (package-vc-install aktiviert nicht sofort → Kompilierungsfehler bei Abhängigkeiten)
(use-package quelpa
  :ensure t
  :custom
  (quelpa-update-melpa-p nil)
  (quelpa-self-upgrade-p nil))

;; websocket (MELPA) — Abhängigkeit von websocket-bridge
(use-package websocket
  :ensure t)

;; websocket-bridge zuerst installieren (Abhängigkeit von org-reminders)
(unless (package-installed-p 'websocket-bridge)
  (quelpa '(websocket-bridge :fetcher github :repo "ginqi7/websocket-bridge")))

;; org-reminders installieren
(unless (package-installed-p 'org-reminders)
  (quelpa '(org-reminders :fetcher github :repo "ginqi7/org-reminders")))

(use-package org-reminders
  :ensure nil
  :custom
  (org-reminders-sync-file (expand-file-name "~/org/reminders.org"))
  ;; "all" | "incomplete" | "complete"
  (org-reminders-display-options "all")
  ;; Anzahl Saves bis ein Sync ausgelöst wird (1 = bei jedem Save)
  (org-reminders-sync-frequency 1))

;; Auto-Sync starten sobald CLI verfügbar (idle damit Startup nicht blockiert)
(when (executable-find "org-reminders")
  (run-with-idle-timer 3 nil
                       (lambda ()
                         (org-reminders-check-sync-file)
                         (org-reminders-start-auto-sync))))

;; Flag: t während package Apple→Org-Updates einspielt → verhindert Sync-Loop
;; Muss defvar sein (special variable) damit dynamisches Scoping durch alle Hook-Aufrufe gilt
(defvar my/org-reminders--from-apple nil)

(with-eval-after-load 'org-reminders
  ;; org-reminders-message-handle wrappen (nicht nur item-update-detail):
  ;; org-reminders-with-sync-file ruft save-buffer NACH dem return von item-update-detail auf,
  ;; deshalb muss das Flag den gesamten message-handle-Aufruf umfassen.
  (advice-add 'org-reminders-message-handle :around
              (lambda (orig &rest args)
                (let ((my/org-reminders--from-apple t))
                  (apply orig args))))
  ;; check-sync-file beim Start unterdrücken (ruft ebenfalls save-buffer auf)
  (advice-add 'org-reminders-check-sync-file :around
              (lambda (orig &rest args)
                (let ((my/org-reminders--from-apple t))
                  (apply orig args)))))

;; LAST-MODIFIED aktualisieren wenn User TODO-State ändert → syncOnce erkennt Änderung
;; (nicht bei package-internem Apple→Org-Sync → würde sonst Loop verursachen)
(defun my/org-reminders-mark-entry-modified ()
  (when (and (not my/org-reminders--from-apple)
             (buffer-file-name)
             (string= (buffer-file-name)
                      (expand-file-name "~/org/reminders.org"))
             (org-entry-get nil "EXTERNAL-ID"))
    (org-entry-put nil "LAST-MODIFIED"
                   (format-time-string "%Y-%m-%d %H:%M:%S"))))

(add-hook 'org-after-todo-state-change-hook #'my/org-reminders-mark-entry-modified)

;; Sync-once nach User-Save — übersprungen wenn package-internes Save (from-apple = t)
;; condition-case fängt websocket-closed ab falls CLI nicht verbunden ist
(defvar my/org-reminders--sync-timer nil)

(defun my/org-reminders-on-save ()
  (when (and (not my/org-reminders--from-apple)
             (buffer-file-name)
             (string= (buffer-file-name)
                      (expand-file-name "~/org/reminders.org")))
    (when my/org-reminders--sync-timer
      (cancel-timer my/org-reminders--sync-timer))
    (setq my/org-reminders--sync-timer
          (run-with-timer 1 nil
                          (lambda ()
                            (condition-case err
                                (org-reminders-send-message "sync-once")
                              (error
                               (message "org-reminders sync-once: %s"
                                        (error-message-string err)))))))))

(add-hook 'after-save-hook #'my/org-reminders-on-save)

;; Auto-Save nach 3s Idle → kein manuelles C-x C-s nötig
(defun my/org-reminders-enable-autosave ()
  (when (and (buffer-file-name)
             (string= (buffer-file-name)
                      (expand-file-name "~/org/reminders.org")))
    (setq-local auto-save-visited-interval 3)
    (auto-save-visited-mode 1)))

(add-hook 'find-file-hook #'my/org-reminders-enable-autosave)
#+end_src
