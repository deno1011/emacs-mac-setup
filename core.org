;; Install a brew package in the background and log it for clean uninstall.
;; TYPE is "formula" or "cask", PACKAGE is the brew package name, BREW-ARGS are
;; the full brew arguments (e.g. "install" "--cask" "font-jetbrains-mono").
(defun my/brew-install-and-log (type package &rest brew-args)
  (let* ((log-file (expand-file-name "system-packages.log" user-emacs-directory))
         (entry    (format "%s:%s\n" type package))
         (proc     (apply #'start-process
                          (format "install-%s" package)
                          (format "*install-%s*" package)
                          "brew" brew-args)))
    (set-process-sentinel
     proc
     (lambda (p _e)
       (when (= (process-exit-status p) 0)
         (unless (and (file-exists-p log-file)
                      (with-temp-buffer
                        (insert-file-contents log-file)
                        (search-forward entry nil t)))
           (append-to-file entry nil log-file)))))))

;; Install a pip package in the background and log it for clean uninstall.
(defun my/pip-install-and-log (package)
  (let* ((log-file (expand-file-name "system-packages.log" user-emacs-directory))
         (entry    (format "pip:%s\n" package))
         (proc     (start-process (format "install-%s" package)
                                  (format "*install-%s*" package)
                                  "pip3" "install" "--break-system-packages" package)))
    (set-process-sentinel
     proc
     (lambda (p _e)
       (when (= (process-exit-status p) 0)
         (unless (and (file-exists-p log-file)
                      (with-temp-buffer
                        (insert-file-contents log-file)
                        (search-forward entry nil t)))
           (append-to-file entry nil log-file)))))))

#+begin_src emacs-lisp :tangle yes
(defun my/config-org-write-guard ()
  "Abort any save that would overwrite a protected config file."
  (when (and buffer-file-name
             (my/gptel-protected-p (file-truename buffer-file-name)))
    (error "Save blocked: %s is a protected config file (buffer: %s)"
           buffer-file-name (buffer-name))))

(add-hook 'before-save-hook #'my/config-org-write-guard)

;; Block write-file targeting protected paths
(define-advice write-file (:before (filename &rest _) gptel-protect)
  (when (and filename
             (my/gptel-protected-p (expand-file-name filename)))
    (error "write-file blocked: %s is a protected config file" filename)))

;; Block linking any buffer to a protected file path
(define-advice set-visited-file-name (:before (filename &rest _) gptel-protect)
  (when (and filename
             (not (string-empty-p filename))
             (my/gptel-protected-p (expand-file-name filename)))
    (error "set-visited-file-name blocked: %s is a protected config file" filename)))

* Startup Sync
#+begin_src emacs-lisp
(defun my/org-encrypted-p (org-dir)
  "Return t if any .org file in ORG-DIR starts with a git-crypt null byte."
  (let ((first (car (and (file-directory-p org-dir)
                         (directory-files org-dir t "\\.org$")))))
    (when first
      (with-temp-buffer
        (insert-file-contents-literally first nil 0 10)
        (string-match-p "\x00" (buffer-string))))))

(defun my/repo-dirty-p (repo)
  "Return t if REPO has uncommitted changes."
  (not (string-empty-p
        (shell-command-to-string
         (format "git -C %s status --porcelain" (shell-quote-argument repo))))))

(defun my/revert-org-buffers ()
  (dolist (buf (buffer-list))
    (with-current-buffer buf
      (when (and (derived-mode-p 'org-mode)
                 buffer-file-name
                 (file-exists-p buffer-file-name))
        (revert-buffer t t t)))))

(defun my/beorg-sync (org-d beorg)
  (when (and (file-directory-p org-d) (file-directory-p beorg))
    (start-process "beorg-sync" " *beorg-sync*"
                   "rsync" "-a" "--delete"
                   org-d (file-name-as-directory beorg))))

(defun my/startup-sync ()
  "Pull emacs-config from origin on startup; unlock git-crypt if needed; sync org/ to beorg."
  (let* ((repo  (file-truename (expand-file-name "~/emacs-config")))
         (org-d (file-name-as-directory (expand-file-name "org" repo)))
         (beorg (expand-file-name
                 "Library/Mobile Documents/iCloud~com~appsonthemove~beorg/Documents/org"
                 (getenv "HOME"))))
    (when (and (file-directory-p repo)
               (file-directory-p (expand-file-name ".git" repo)))
      (let ((proc (let ((default-directory repo))
                    (start-process "startup-sync" " *startup-sync*"
                                   "git" "pull" "--ff-only" "origin" "main"))))
        (set-process-sentinel
         proc
         (lambda (p _)
           (when (= (process-exit-status p) 0)
             (message "Startup sync: pull complete.")
             (if (not (my/org-encrypted-p org-d))
                 (progn (my/revert-org-buffers)
                        (my/beorg-sync org-d beorg))
               ;; Org files are encrypted — git-crypt refuses to unlock when
               ;; the working tree is dirty. Surgically reset only the
               ;; encrypted paths (org/, emacs.d/secrets.el) and the
               ;; auto-generated config.el. Real edits (config.org) are
               ;; left untouched; if they still block unlock, warn the user.
               (when (my/repo-dirty-p repo)
                 (call-process "git" nil nil nil "-C" repo
                               "checkout" "--" "org/" "emacs.d/secrets.el" "config.el")
                 (if (my/repo-dirty-p repo)
                     (display-warning 'emacs-setup
                      (concat "Uncommitted changes block git-crypt unlock.\n"
                              "Commit or stash them, then restart Emacs.\n"
                              "Or run manually: bash ~/unlock-git-crypt.sh")
                      :warning)
                   (message "Startup sync: reset encrypted blobs before unlock.")))
               (unless (my/repo-dirty-p repo)
               (message "Startup sync: org files encrypted — running unlock-git-crypt.sh...")
               (let ((unlock (start-process "git-crypt-unlock" " *git-crypt-unlock*"
                                            (expand-file-name "~/unlock-git-crypt.sh"))))
                 (set-process-sentinel
                  unlock
                  (lambda (u _)
                    (if (= (process-exit-status u) 0)
                        (progn
                          (message "Startup sync: git-crypt unlocked.")
                          (my/revert-org-buffers)
                          (my/beorg-sync org-d beorg))
                      (display-warning 'emacs-setup
                       "git-crypt unlock failed — run: bash ~/unlock-git-crypt.sh"
                       :warning))))))))))))))

(add-hook 'after-init-hook #'my/startup-sync)

  "Convert selected region to meeting notes format and copy to kill ring."
  (interactive)
  (let (prefix)
    (save-excursion
      (save-restriction
        (narrow-to-region (region-beginning) (region-end))
        (untabify (point-min) (point-max))
        (goto-char (point-min))
