#+TITLE: Emacs Configuration
#+STARTUP: overview

Loaded by: [[file:~/.emacs.d/init.el][init.el]]

Configuration is split across three domain files. This file bootstraps them —
downloading any that are missing on first install from the bundled defaults in
[[https://github.com/deno1011/emacs-mac-setup][emacs-mac-setup]].

- [[file:core.org][core.org]] — Emacs base: packages, UI, font, version control, protected files, startup sync
- [[file:org-setup.org][org-setup.org]] — Org mode: agenda, capture, tags, clocking, export, LaTeX, graphics
- [[file:gptel-setup.org][gptel-setup.org]] — AI: gptel backends, tools, Claude Executor, workspace context, system prompt
- [[file:apple-reminders.org][apple-reminders.org]] — Apple Reminders (JXA/osascript, kein CLI) — /nur macOS, nicht docker/orbstack/
- ~~[[file:org-reminders-setup.org][org-reminders-setup.org]]~~ — /deaktiviert/ — [[https://github.com/ginqi7/org-reminders][ginqi7/org-reminders]] zu fehlerhaft für produktiven Einsatz ([[https://github.com/ginqi7/org-reminders/issues][Issues]])

To apply changes after editing a sub-file, run =M-x org-babel-tangle= in that file,
then =M-x load-file= on the generated =.el= file (or restart Emacs).

* Bootstrap

#+begin_src elisp :tangle yes
;; org-reminders-setup.org deactivated — ginqi7/org-reminders too buggy for production use
;; (sync unreliable, Apple→Emacs overwrites Emacs changes, priority handling broken)
;; See: https://github.com/ginqi7/org-reminders/issues
;; Replaced by apple-reminders.org — custom JXA/osascript implementation, no CLI needed
(defvar my/config-split-files '("core.org" "org-setup.org" "gptel-setup.org"))
(when (eq system-type 'darwin)
  (add-to-list 'my/config-split-files "apple-reminders.org" t))

(defvar my/config-fallback-base-url
  "https://raw.githubusercontent.com/deno1011/emacs-mac-setup/stable/"
  "Base URL for fetching missing config files on first install.")

(defun my/ensure-config-file (filename)
  (let ((path (expand-file-name filename my/config-dir)))
    (unless (file-exists-p path)
      (message "Config: fetching missing %s from emacs-mac-setup..." filename)
      (condition-case err
          (progn (require 'url)
                 (url-copy-file (concat my/config-fallback-base-url filename) path t))
        (error (message "Config: could not fetch %s: %s" filename err))))
    path))

(dolist (f my/config-split-files)
  (let ((path (my/ensure-config-file f)))
    (when (file-exists-p path)
      (condition-case err
          (org-babel-load-file path)
        (error (message "CONFIG LOAD ERROR (%s): %s" f err))))))
#+end_src
