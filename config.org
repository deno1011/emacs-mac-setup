;;; -*- lexical-binding: t -*-
(message "Loading config... (first start may take a few minutes while packages install)")

;; Install a brew package in the background and log it for clean uninstall.
;; TYPE is "formula" or "cask", PACKAGE is the brew package name, BREW-ARGS are
;; the full brew arguments (e.g. "install" "--cask" "font-jetbrains-mono").
(defun my/brew-install-and-log (type package &rest brew-args)
  (let* ((log-file (expand-file-name "system-packages.log" user-emacs-directory))
         (entry    (format "%s:%s\n" type package))
         (proc     (apply #'start-process
                          (format "install-%s" package)
                          (format "*install-%s*" package)
                          "brew" brew-args)))
    (set-process-sentinel
     proc
     (lambda (p _e)
       (when (= (process-exit-status p) 0)
         (unless (and (file-exists-p log-file)
                      (with-temp-buffer
                        (insert-file-contents log-file)
                        (search-forward entry nil t)))
           (append-to-file entry nil log-file)))))))

* Startup Sync
#+begin_src emacs-lisp
(defun my/startup-sync ()
  "Pull emacs-config from origin on startup; sync org/ to beorg on success."
  (let* ((repo  (file-truename (expand-file-name "~/emacs-config")))
         (org-d (file-name-as-directory (expand-file-name "org" repo)))
         (beorg (expand-file-name
                 "Library/Mobile Documents/iCloud~com~appsonthemove~beorg/Documents/org"
                 (getenv "HOME"))))
    (when (and (file-directory-p repo)
               (file-directory-p (expand-file-name ".git" repo)))
      (let ((proc (let ((default-directory repo))
                    (start-process "startup-sync" " *startup-sync*"
                                   "git" "pull" "--ff-only" "origin" "main"))))
        (set-process-sentinel
         proc
         (lambda (p _)
           (when (= (process-exit-status p) 0)
             (message "Startup sync: pull complete.")
             (org-revert-all-org-buffers)
             (when (and (file-directory-p org-d) (file-directory-p beorg))
               (start-process "beorg-sync" " *beorg-sync*"
                              "rsync" "-a" "--delete"
                              org-d (file-name-as-directory beorg))))))))))

(add-hook 'after-init-hook #'my/startup-sync)

* Project & Task Detection
#+begin_src emacs-lisp
(defun bh/is-project-p ()
  "Any task with a todo keyword subtask."
  (save-restriction
    (widen)
    (let ((has-subtask)
          (subtree-end (save-excursion (org-end-of-subtree t)))
          (is-a-task (member (nth 2 (org-heading-components)) org-todo-keywords-1)))
      (save-excursion
        (forward-line 1)
        (while (and (not has-subtask)
                    (< (point) subtree-end)
                    (re-search-forward "^\*+ " subtree-end t))
          (when (member (org-get-todo-state) org-todo-keywords-1)
            (setq has-subtask t))))
      (and is-a-task has-subtask))))

(defun bh/is-project-subtree-p ()
  "Any task in a project subtree."
  (let ((task (save-excursion (org-back-to-heading 'invisible-ok) (point))))
    (save-excursion
      (bh/find-project-task)
      (if (equal (point) task) nil t))))

(defun bh/is-task-p ()
  "Any task with a todo keyword and no subtask."
  (save-restriction
    (widen)
    (let ((has-subtask)
          (subtree-end (save-excursion (org-end-of-subtree t)))
          (is-a-task (member (nth 2 (org-heading-components)) org-todo-keywords-1)))
      (save-excursion
        (forward-line 1)
        (while (and (not has-subtask)
                    (< (point) subtree-end)
                    (re-search-forward "^\*+ " subtree-end t))
          (when (member (org-get-todo-state) org-todo-keywords-1)
            (setq has-subtask t))))
      (and is-a-task (not has-subtask)))))

(defun bh/is-subproject-p ()
  "Any task which is a subtask of another project."
  (let ((is-subproject)
        (is-a-task (member (nth 2 (org-heading-components)) org-todo-keywords-1)))
    (save-excursion
      (while (and (not is-subproject) (org-up-heading-safe))
        (when (member (nth 2 (org-heading-components)) org-todo-keywords-1)
          (setq is-subproject t))))
    (and is-a-task is-subproject)))

(defun bh/mark-next-parent-tasks-todo ()
  "Visit each parent task and change NEXT states to TODO."
  (let ((mystate (or (and (fboundp 'org-state) state)
                     (nth 2 (org-heading-components)))))
    (when mystate
      (save-excursion
        (while (org-up-heading-safe)
          (when (member (nth 2 (org-heading-components)) (list "NEXT"))
            (org-todo "TODO")))))))

(add-hook 'org-after-todo-state-change-hook 'bh/mark-next-parent-tasks-todo 'append)
(add-hook 'org-clock-in-hook 'bh/mark-next-parent-tasks-todo 'append)

(defvar bh/hide-scheduled-and-waiting-next-tasks t)

(defun bh/list-sublevels-for-projects-indented ()
  (if (marker-buffer org-agenda-restrict-begin)
      (setq org-tags-match-list-sublevels 'indented)
    (setq org-tags-match-list-sublevels nil))

(defun bh/list-sublevels-for-projects ()
  (if (marker-buffer org-agenda-restrict-begin)
      (setq org-tags-match-list-sublevels t)
    (setq org-tags-match-list-sublevels nil))
  nil)

(defun bh/skip-stuck-projects ()
  (save-restriction
    (widen)
    (let ((next-headline (save-excursion (or (outline-next-heading) (point-max)))))
      (if (bh/is-project-p)
          (let* ((subtree-end (save-excursion (org-end-of-subtree t)))
                 (has-next))
            (save-excursion
              (forward-line 1)
              (while (and (not has-next) (< (point) subtree-end)
                          (re-search-forward "^\\*+ NEXT " subtree-end t))
                (unless (member "WAITING" (org-get-tags-at))
                  (setq has-next t))))
            (if has-next nil next-headline))
        nil))))

(defun bh/skip-non-stuck-projects ()
  (save-restriction
    (widen)
    (let ((next-headline (save-excursion (or (outline-next-heading) (point-max)))))
      (if (bh/is-project-p)
          (let* ((subtree-end (save-excursion (org-end-of-subtree t)))
                 (has-next))
            (save-excursion
              (forward-line 1)
              (while (and (not has-next) (< (point) subtree-end)
                          (re-search-forward "^\\*+ NEXT " subtree-end t))
                (unless (member "WAITING" (org-get-tags-at))
                  (setq has-next t))))
            (if has-next next-headline nil))
        next-headline))))

(defun bh/skip-non-projects ()
  (if (save-excursion (bh/skip-non-stuck-projects))
      (save-restriction
        (widen)
        (let ((subtree-end (save-excursion (org-end-of-subtree t))))
          (cond ((bh/is-project-p) nil)
                ((and (bh/is-project-subtree-p) (not (bh/is-task-p))) nil)
                (t subtree-end))))
    (save-excursion (org-end-of-subtree t))))

(defun bh/skip-non-tasks ()
  (save-restriction
    (widen)
    (let ((next-headline (save-excursion (or (outline-next-heading) (point-max)))))
      (cond ((bh/is-task-p) nil)
            (t next-headline)))))

(defun bh/skip-project-trees-and-habits ()
  (save-restriction
    (widen)
    (let ((subtree-end (save-excursion (org-end-of-subtree t))))
      (cond ((bh/is-project-p) subtree-end)
            ((org-is-habit-p) subtree-end)
            (t nil)))))

(defun bh/skip-projects-and-habits-and-single-tasks ()
  (save-restriction
    (widen)
    (let ((next-headline (save-excursion (or (outline-next-heading) (point-max)))))
      (cond ((org-is-habit-p) next-headline)
            ((and bh/hide-scheduled-and-waiting-next-tasks
                  (member "WAITING" (org-get-tags-at))) next-headline)
            ((bh/is-project-p) next-headline)
            ((and (bh/is-task-p) (not (bh/is-project-subtree-p))) next-headline)
            (t nil)))))

(defun bh/skip-project-tasks-maybe ()
  (save-restriction
    (widen)
    (let* ((subtree-end (save-excursion (org-end-of-subtree t)))
           (next-headline (save-excursion (or (outline-next-heading) (point-max))))
           (limit-to-project (marker-buffer org-agenda-restrict-begin)))
      (cond ((bh/is-project-p) next-headline)
            ((org-is-habit-p) subtree-end)
            ((and (not limit-to-project) (bh/is-project-subtree-p)) subtree-end)
            ((and limit-to-project (bh/is-project-subtree-p)
                  (member (org-get-todo-state) (list "NEXT"))) subtree-end)
            (t nil)))))

(defun bh/skip-project-tasks ()
  (save-restriction
    (widen)
    (let* ((subtree-end (save-excursion (org-end-of-subtree t))))
      (cond ((bh/is-project-p) subtree-end)

(defun bh/skip-non-project-tasks ()
  (save-restriction
    (widen)
    (let* ((subtree-end (save-excursion (org-end-of-subtree t)))
           (next-headline (save-excursion (or (outline-next-heading) (point-max)))))

(defun bh/skip-projects-and-habits ()
  (save-restriction
    (widen)

(defmacro bh/agenda-sort-test (fn a b)
  `(cond ((and (apply ,fn (list ,a)) (apply ,fn (list ,b))) (setq result nil))
         ((apply ,fn (list ,a)) (setq result -1))
         ((apply ,fn (list ,b)) (setq result 1))
         (t nil)))

(defmacro bh/agenda-sort-test-num (fn compfn a b)
  `(cond ((apply ,fn (list ,a))
          (setq num-a (string-to-number (match-string 1 ,a)))
          (if (apply ,fn (list ,b))
              (progn (setq num-b (string-to-number (match-string 1 ,b)))
                     (setq result (if (apply ,compfn (list num-a num-b)) -1 1)))
            (setq result -1)))
         ((apply ,fn (list ,b)) (setq result 1))
         (t nil)))

          ((bh/agenda-sort-test-num 'bh/is-scheduled-late '> a b))
          ((bh/agenda-sort-test-num 'bh/is-pending-deadline '< a b))
          (t (setq result nil)))
    result))

* Claude Executor

Verbindet [[https://github.com/karthink/gptel][gptel]] mit Auto-Ausführungs-Hooks, sodass Claude direkt
Aktionen in Emacs auslösen kann (Agenda öffnen, TODOs einfügen,
Fenster verwalten, Dateien öffnen, …) — und beliebige Code-Blöcke
in Antworten ausführen kann.

** Übersicht

*** Drei Ausführungs-Mechanismen

1. *AUTORUN-Elisp-Blöcke* — Hauptmechanismus für Emacs-Steuerung.
   Claude antwortet mit ~#+begin_src elisp :AUTORUN ... #+end_src~.
   Der Hook findet, evaluiert und loggt das Ergebnis.
2. *Exec-Tags* — ~<!-- exec: KOMMANDO -->~ in Antworten triggert
   einzelne Shell-Aufrufe (mit Bestätigung).
3. *Babel-Code-Blöcke* — Klassische ~#+begin_src python/sh/R/...~
   Blöcke werden in Temp-Files geschrieben und über die jeweilige
   Sprach-CLI ausgeführt.

Zusätzlich: *Auto-Pattern-Matching* via ~claude-executor-auto-commands~
(Regex auf Antworttext → Shell-Kommando).

*** Ablauf einer Antwort

#+begin_example
gptel-Antwort fertig
        │
        ▼
gptel-post-response-functions
        │
        ├── claude-executor-response-hook
        │       wenn auto-execute = t:
        │       1. alle Babel-Blöcke ausführen (Shell)
        │       2. exec-Tags ausführen (mit Bestätigung)
        │       3. auto-commands Pattern matchen
        │
        └── claude-executor-autorun-hook
                immer aktiv solange Mode an:
                Findet :AUTORUN Elisp-Blöcke
                → Whitelist-Check (optional)
                → safe-eval mit Logging in *Claude Actions*
#+end_example

*** Sicherheit

| Schalter                                  | Default | Wirkung                                           |
|-------------------------------------------+---------+---------------------------------------------------|
| ~claude-executor-auto-execute~            | nil     | aus → Babel + exec-Tags + auto-commands inaktiv   |
| ~claude-executor-confirm-before-execute~  | nil     | wenn t → Bestätigungs-Dialog vor Babel & AUTORUN  |
| ~claude-executor-auto-execute-elisp~      | nil     | reserviert für künftige Elisp-Auto-Eval-Steuerung |
| ~claude-executor-allowed-functions~       | nil     | nil = alles erlaubt; sonst Whitelist für AUTORUN  |
| ~claude-executor-mode~                    | t       | zentraler An/Aus-Schalter für *beide* Hooks       |

** Babel-Block-Ausführung

Findet ~#+begin_src LANG ... #+end_src~ in der Antwort, schreibt den
Body in eine Temp-Datei, ruft die jeweilige Sprach-CLI auf und loggt
in den Buffer ~*Claude Execution Results*~.

*** Block finden

#+begin_src elisp :tangle yes
(defun claude-executor--find-babel-blocks ()
  "Findet alle Babel Code-Blöcke in der aktuellen (ggf. eingeschränkten) Region.
Gibt eine Liste von Plists mit :lang, :header, :body, :begin, :end zurück."
  (let (blocks)
    (save-excursion
      (goto-char (point-min))
      (while (re-search-forward
              "^#\\+begin_src\\s-+\\(\\w+\\)\\(.*\\)\n\\(\\(?:.*\n\\)*?\\)#\\+end_src"
              nil t)
        (push (list :lang   (match-string-no-properties 1)
                    :header (match-string-no-properties 2)
                    :body   (match-string-no-properties 3)
                    :begin  (match-beginning 0)
                    :end    (match-end 0))
              blocks)))
    (nreverse blocks)))
#+end_src

*** Ausführen & Loggen

#+begin_src elisp :tangle yes
(defun claude-executor--do-execute (lang body)
  "Schreibt BODY in Temp-Datei, ruft passende Sprach-CLI auf,
loggt in *Claude Execution Results* und löscht die Temp-Datei."
  (let* ((ext        (or (claude-executor--lang-extension lang) ""))
         (tmp-file   (make-temp-file "claude-exec-" nil ext))
         (result-buf (get-buffer-create "*Claude Execution Results*")))
    (with-temp-file tmp-file
      (insert body))
    (with-current-buffer result-buf
      (goto-char (point-max))
      (insert (format "\n=== %s [%s] ===\n" lang
                      (format-time-string "%H:%M:%S")))
      (insert body)
      (insert "\n--- Output ---\n")
      (let ((cmd (claude-executor--lang-command lang tmp-file)))
        (if cmd
            (insert (shell-command-to-string cmd))
          (insert (format "Keine Ausführung für Sprache: %s\n" lang))))
      (insert "\n"))
    (display-buffer result-buf)
    (delete-file tmp-file)))

(defun claude-executor--execute-babel-block (block)
  "Führt einen Babel-Block aus.
Prüft `claude-executor-allowed-languages' und ggf.
`claude-executor-confirm-before-execute'."
  (let ((lang (plist-get block :lang))
        (body (plist-get block :body)))
    (when (member lang claude-executor-allowed-languages)
      (if claude-executor-confirm-before-execute
          (when (yes-or-no-p
                 (format "Block ausführen (%s):\n%s\n?" lang body))
            (claude-executor--do-execute lang body))
        (claude-executor--do-execute lang body)))))
#+end_src

** Exec-Tag-Ausführung

Erlaubt einzelne Shell-Kommandos via HTML-Kommentar in der Antwort:
~<!-- exec: ls -la ~/projects -->~. Wird mit Bestätigung ausgeführt.

#+begin_src elisp :tangle yes
(defun claude-executor--find-commands ()
  "Findet Kommandos in Tags der Form '<!-- exec: KOMMANDO -->'."
  (let (commands)
    (save-excursion
      (goto-char (point-min))
      (while (re-search-forward
              "<!--\\s-*exec:\\s-*\\(.*?\\)\\s-*-->" nil t)
        (push (match-string-no-properties 1) commands)))
    (nreverse commands)))

(defun claude-executor-run-command (command)
  "Führt COMMAND in der Shell aus.
Output landet im Buffer *Claude Commands*.
Auch interaktiv aufrufbar: M-x oder C-c C-x c."
  (interactive "sKommando: ")
  (let ((result-buf (get-buffer-create "*Claude Commands*")))
    (with-current-buffer result-buf
      (goto-char (point-max))
      (insert (format "\n$ %s\n" command))
      (insert (shell-command-to-string command))
      (insert "\n"))
    (display-buffer result-buf)))
#+end_src

** AUTORUN-Elisp-Blöcke

Hauptmechanismus für die Emacs-Steuerung. Claude markiert ausführbare
Elisp-Blöcke mit dem Tag aus ~claude-executor-elisp-tag~ (default
"AUTORUN"). Der Hook findet sie, prüft optional gegen die Whitelist,
evaluiert sie unter Error-Handling und protokolliert Ergebnis/Fehler
in ~*Claude Actions*~.

*** Block finden

#+begin_src elisp :tangle yes
(defun claude-executor--find-autorun-blocks ()
  "Findet Elisp-Blöcke deren Header das AUTORUN-Tag enthält.
Suche ist abhängig von `claude-executor-elisp-tag'."
  (let (blocks)
    (save-excursion
      (goto-char (point-min))
      (while (re-search-forward
              (format "^#\\+begin_src elisp.*%s.*\n\\(\\(?:.*\n\\)*?\\)#\\+end_src"
                      claude-executor-elisp-tag)
              nil t)
        (push (match-string-no-properties 1) blocks)))
    (nreverse blocks)))
#+end_src

*** Whitelist-Prüfung

Optional: nur Aufrufe von Funktionen aus ~claude-executor-allowed-functions~
zulassen. Wenn die Liste leer ist, ist alles erlaubt.

#+begin_src elisp :tangle yes
(defun claude-executor--whitelist-allows-p (code)
  "Prüft ob CODE (String) nur Funktionen aus
`claude-executor-allowed-functions' aufruft.
Wenn keine Whitelist gesetzt ist → t (alles erlaubt)."
  (if (null claude-executor-allowed-functions)
      t
    (let ((sexp    (read code))
          (allowed claude-executor-allowed-functions)
          (ok      t))
      (cl-labels ((walk (form)
                    (cond
                     ((and (consp form) (symbolp (car form)))
                      (unless (memq (car form) allowed)
                        (setq ok nil))
                      (mapc #'walk (cdr form)))
                     ((consp form)
                      (mapc #'walk form)))))
        (walk sexp))
      ok)))
#+end_src

*** Safe-Eval mit Logging

#+begin_src elisp :tangle yes
(defun claude-executor--safe-eval (code)
  "Evaluiert CODE (String) mit Error-Handling und Logging.
- Whitelist-Verletzung → ⚠ Eintrag, kein Eval.
- Erfolg               → ✓ Eintrag mit Result.
- Fehler               → ✗ Eintrag mit Error-Message.
Logs landen im Buffer *Claude Actions*."
  (let ((log-buf (get-buffer-create "*Claude Actions*")))
    (cond
     ((not (claude-executor--whitelist-allows-p code))
      (with-current-buffer log-buf
        (goto-char (point-max))
        (insert (format "\n⚠ [%s]\n  Code: %s\n  Abgewiesen: nicht in Whitelist.\n"
                        (format-time-string "%H:%M:%S") code))
        (display-buffer log-buf)))
     (t
      (condition-case err
          (let ((result (eval (read code))))
            (with-current-buffer log-buf
              (goto-char (point-max))
              (insert (format "\n✓ [%s]\n  Code: %s\n  Result: %s\n"
                              (format-time-string "%H:%M:%S")
                              code result))
              (display-buffer log-buf))
            result)
        (error
         (with-current-buffer log-buf
           (goto-char (point-max))
           (insert (format "\n✗ [%s]\n  Code: %s\n  Error: %s\n"
                           (format-time-string "%H:%M:%S")
                           code err))
           (display-buffer log-buf))))))))
#+end_src

*** Response-Hook (Babel + exec-Tags + auto-commands)

#+begin_src elisp :tangle yes
(defun claude-executor-response-hook (beg end)
  "Hook für Babel-Blöcke, exec-Tags und Auto-Pattern.
Aktiv nur wenn `claude-executor-auto-execute' = t.
Wird in `gptel-post-response-functions' registriert."
  (when claude-executor-auto-execute
    (save-restriction
      (narrow-to-region beg end)
      ;; 1. Babel-Blöcke
      (dolist (block (claude-executor--find-babel-blocks))
        (claude-executor--execute-babel-block block))
      ;; 2. Explizite exec-Tags
      (dolist (cmd (claude-executor--find-commands))
        (when (yes-or-no-p (format "Kommando ausführen: %s ?" cmd))
          (claude-executor-run-command cmd)))
      ;; 3. User-definierte Pattern aus claude-executor-auto-commands
      (dolist (pattern-cmd claude-executor-auto-commands)
        (save-excursion
          (goto-char (point-min))
          (when (re-search-forward (car pattern-cmd) nil t)
            (claude-executor-run-command (cdr pattern-cmd))))))))
#+end_src

      (message "Keine Code-Blöcke gefunden."))))

(defun claude-executor-execute-block-at-point ()
  "Führt den Babel-Block unter dem Cursor aus."
  (interactive)
  (let* ((elem (org-element-at-point))
         (lang (org-element-property :language elem))
         (body (org-element-property :value elem)))
    (if (and lang body)
        (claude-executor--do-execute lang body)
      (message "Kein Code-Block unter Cursor."))))
#+end_src

** Minor Mode

Zentraler An/Aus-Schalter. Registriert *beide* Hooks (Response +
AUTORUN) an ~gptel-post-response-functions~, damit
~(claude-executor-mode 0)~ wirklich alles abschaltet.

#+begin_src elisp :tangle yes
(defvar claude-executor-mode-map
  (let ((map (make-sparse-keymap)))
    (define-key map (kbd "C-c C-x e") #'claude-executor-execute-all-blocks)
    (define-key map (kbd "C-c C-x b") #'claude-executor-execute-block-at-point)
    (define-key map (kbd "C-c C-x c") #'claude-executor-run-command)
    map)
  "Keymap für `claude-executor-mode'.")

(define-minor-mode claude-executor-mode
  "Globaler Mode für automatische Ausführung von Claude-Antworten.
Aktiviert die Response- und AUTORUN-Hooks an
`gptel-post-response-functions'."
  :lighter " CE"
  :keymap claude-executor-mode-map
  :global t
  (if claude-executor-mode
      (progn
        (add-hook 'gptel-post-response-functions
                  #'claude-executor-response-hook)
        (add-hook 'gptel-post-response-functions
                  #'claude-executor-autorun-hook)
        (message "Claude Executor aktiviert."))
    (remove-hook 'gptel-post-response-functions
                 #'claude-executor-response-hook)
    (remove-hook 'gptel-post-response-functions
                 #'claude-executor-autorun-hook)
    (message "Claude Executor deaktiviert.")))

(provide 'claude-executor)
#+end_src

** Keybindings Übersicht

|---------------+-----------------------------------------|

| ~C-c C-x b~   | Babel-Block unter Cursor ausführen      |

** Beispiele & Workflows

*** AUTORUN: TODO einfügen

Im gptel-Buffer:

#+begin_example
Füge "Milch kaufen" als TODO in inbox.org ein
#+end_example

Claude antwortet mit einem AUTORUN-Block:

#+begin_example
,#+begin_src elisp :AUTORUN
(with-current-buffer (find-file-noselect "~/org/inbox.org")
  (goto-char (point-max))
  (insert "\n* TODO Milch kaufen")
  (save-buffer))
,#+end_src
#+end_example

→ Hook erkennt =AUTORUN= → (ggf. Bestätigung) → Eval → Eintrag in
~*Claude Actions*~.

*** AUTORUN: Agenda öffnen

Du fragst: „Zeig mir die Agenda für diese Woche".

Claude antwortet:

#+begin_example
,#+begin_src elisp :AUTORUN
(org-agenda nil "a")
,#+end_src
#+end_example

*** Exec-Tag: Shell-Kommando

Claude antwortet mit:

#+begin_example
<!-- exec: ls -la ~/projects -->
#+end_example

→ Wird (mit Bestätigung) ausgeführt, Output in ~*Claude Commands*~.
Voraussetzung: ~claude-executor-auto-execute~ = t.

*** Babel: Python-Snippet ausführen

Antwort enthält:

#+begin_example
,#+begin_src python
print("hello")
,#+end_src
#+end_example

→ Bei ~claude-executor-auto-execute~ = t: Temp-File → ~python3~ →
Output in ~*Claude Execution Results*~.
