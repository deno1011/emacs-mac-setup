This setup file intentionally stays small. The reusable AI implementation lives
in the standalone package repository:

[[https://github.com/deno1011/gptel-agent-runtime][deno1011/gptel-agent-runtime]]

The setup repository owns first-install choices: which package branch is used,
which local provider is preferred, and which local model should be ready after a
fresh Mac setup. The runtime package owns agent behavior, tools, memory, routing,
and model-neutral execution.

* Repository Roles

- =~/emacs-mac-setup= is the installer/bootstrap/template repo.
- =~/emacs= or =~/emacs-data= is the private live Emacs runtime repo.
- =~/gptel-agent-runtime= is the standalone package development repo.
- =~/.emacs.d/config-readonly= is the generated readonly load target.

* Environment Pre-flight

Checks if Ollama is running and pulls standard models if missing. This provides
visible progress in the =*gptel-setup-ollama*= buffer.

#+begin_src emacs-lisp :tangle yes
(defun my/gptel-setup--ollama-url (&optional path)
  (format "http://%s%s" my/gptel-setup-ollama-host (or path "")))

(defun my/gptel-setup-ollama-running-p ()
  (condition-case nil
      (let ((buf (url-retrieve-synchronously
                  (my/gptel-setup--ollama-url "/api/tags") t t 1)))
        (when buf (kill-buffer buf) t))
    (error nil)))

(defun my/gptel-setup-ensure-ollama-server ()
  "Start Ollama server if not running."
  (unless (my/gptel-setup-ollama-running-p)
    (if (not (executable-find "ollama"))
        (message "WARN: 'ollama' not found in PATH")
      (message "Starting Ollama server...")
      (let ((process-environment (copy-sequence process-environment)))
        (when my/gptel-setup-ollama-models-directory
          (setenv "OLLAMA_MODELS" (expand-file-name my/gptel-setup-ollama-models-directory)))
        (start-process "ollama-serve" "*ollama-serve*" "ollama" "serve"))
      ;; Wait a bit for server to breathe
      (sleep-for 2))))

(defun my/gptel-setup-pull-models (models)
  "Pull MODELS sequentially with progress in *gptel-setup-ollama*."
  (when models
    (let ((model (car models))
          (remaining (cdr models)))
      (message "Checking Ollama model %s..." model)
      (make-process
       :name "gptel-setup-check"
       :buffer "*gptel-setup-ollama*"
       :command (list "ollama" "list")
       :noquery t
       :sentinel
       (lambda (proc _)
         (when (and (not (process-live-p proc)) (zerop (process-exit-status proc)))
           (with-current-buffer (process-buffer proc)
             (goto-char (point-min))
             (if (re-search-forward (regexp-quote (symbol-name model)) nil t)
                 (progn
                   (message "Ollama model %s is ready." model)
                   (my/gptel-setup-pull-models remaining))
               (message "Ollama model %s missing. Pulling now (see *gptel-setup-ollama*)..." model)
               (display-buffer "*gptel-setup-ollama*")
               (make-process
                :name (format "gptel-pull-%s" model)
                :buffer "*gptel-setup-ollama*"
                :command (list "ollama" "pull" (symbol-name model))
                :noquery t
                :sentinel
                (lambda (pull-proc _)
                  (when (not (process-live-p pull-proc))
                    (if (zerop (process-exit-status pull-proc))
                        (progn
                          (message "Ollama model %s pulled successfully." model)
                          (my/gptel-setup-pull-models remaining))
                      (message "ERROR: Failed to pull %s. Check *gptel-setup-ollama*." model))))))))))))))

(defun my/gptel-setup-pre-flight ()
  "Run environment checks and model provisioning."
  (interactive)
  (when (executable-find "ollama")
    (my/gptel-setup-ensure-ollama-server)
    (my/gptel-setup-pull-models my/gptel-setup-standard-models)))

;; Run pre-flight 5 seconds after idle
(run-with-idle-timer 5 nil #'my/gptel-setup-pre-flight)
#+end_src

* Package Installation and Auto-Update

Installs =gptel-agent-runtime= from GitHub via =package-vc-install= (Emacs 29+).

#+begin_src emacs-lisp :tangle yes
(unless (package-installed-p 'gptel-agent-runtime)
  (package-vc-install
   `(gptel-agent-runtime
     :url "https://github.com/deno1011/gptel-agent-runtime"
     :branch ,my/gptel-agent-package-branch)))

;; Async update check
(when-let* ((lib (locate-library "gptel-agent-runtime"))
            (dir (file-name-directory lib)))
  (run-with-idle-timer
   10 nil
   (lambda (update-dir update-branch)
     (make-process
      :name "gptel-agent-runtime-update"
      :command (list "git" "-C" update-dir "pull" "--rebase" "--quiet"
                     "origin" update-branch)
      :noquery t
      :sentinel (lambda (proc _)
                  (when (and (not (process-live-p proc))
                             (zerop (process-exit-status proc)))
                    (message "gptel-agent-runtime: updated from %s — restart to apply"
                             update-branch)))))
   dir my/gptel-agent-package-branch))
#+end_src

* Runtime Configuration

Registers local backends and selects the default model.

#+begin_src emacs-lisp :tangle yes
(require 'gptel-ollama nil t)

(defun my/gptel-setup-register-backends ()
  "Register local Ollama backends."
  (interactive)
  (when (executable-find "ollama")
    (let ((backend (gptel-make-ollama "Ollama"
                     :stream t
                     :host my/gptel-setup-ollama-host
                     :models '((qwen2.5-coder:7b :capabilities (tool-use json))
                               (qwen2.5:7b-instruct :capabilities (tool-use json))
                               (deepseek-r1 :capabilities (tool-use json))
                               llama3.2 mistral))))
      (setq gptel-backend backend
            gptel-model   my/gptel-setup-default-ollama-model)
      
      ;; Ensure my/gptel-backends is updated for model switching UI
      (setq my/gptel-backends
            (append (cl-remove "Ollama" my/gptel-backends :key #'car :test #'string-prefix-p)
                    `(("Qwen 2.5 Coder 7B (Ollama)" ,backend . qwen2.5-coder:7b)
                      ("Qwen 2.5 Instruct 7B (Ollama)" ,backend . qwen2.5:7b-instruct)
                      ("DeepSeek R1 (Ollama)" ,backend . deepseek-r1))))
      
      (when (fboundp 'my/gptel-sync-directive-for-current-runtime)
        (my/gptel-sync-directive-for-current-runtime))
      (when (fboundp 'my/gptel-sync-tools)
        (my/gptel-sync-tools))
      (message "gptel local backend registered and model %s selected." gptel-model))))

(use-package gptel-agent-runtime
  :ensure nil
  :demand t
  :config
  (my/gptel-setup-register-backends)
  (message "gptel-agent-runtime loaded and local backends configured."))
#+end_src
