#+TITLE: Bootstrap — Bitwarden, GitHub, Personal data, Secrets
#+STARTUP: overview

Loaded *first* by [[file:config.org][config.org]] (prepended to =my/config-split-files=).
Everything here runs before =core.org=, =org-setup.org=, etc., so by
the time the rest of the config loads, API keys are in =setenv=, the
user's private repo is synced into =my/data-dir=, and =gh= is
authenticated.

Flow (idempotent on every Emacs start; silent after first launch):
1. Unlock Bitwarden — prompts on first launch, cached in macOS Keychain after.
2. Read or create the =github-cli-token= BW item (token + identity fields).
3. Install & authenticate =gh= CLI.
4. Sync the user's private repo into =my/data-dir= (clone / pull / create).
5. Generate starter content for any files missing under =my/data-dir=.
6. Load API keys from BW (preferred) or macOS Keychain (fallback).
7. Lock the Bitwarden vault — values stay in Emacs memory; the session
   token is discarded.

Customisation tunables (override in =~/.emacs.d/secrets.el= which loads before this file):
- =my/bw-credentials-item=   :: BW item name holding ALL Emacs secrets (default "emacs-credentials")
- =my/keychain-service=      :: macOS Keychain service for cached BW unlock + API key fallback (default "emacs-credentials")
- =my/bw-legacy-github-item= :: Pre-consolidation BW item ("github-cli-token") — migrated then kept as backup
- =my/bw-legacy-api-item=    :: Pre-consolidation BW item ("emacs-api-keys") — migrated then kept as backup

* Step 1 — macOS Keychain primitives

#+begin_src emacs-lisp :tangle yes
(defun my/keychain-get (service &optional account)
  "Return password stored at SERVICE for ACCOUNT (default user-login-name).
nil if missing or `security' fails."
  (let* ((acc (shell-quote-argument (or account (user-login-name))))
         (svc (shell-quote-argument service))
         (out (with-output-to-string
                (with-current-buffer standard-output
                  (call-process-shell-command
                   (format "security find-generic-password -a %s -s %s -w 2>/dev/null"
                           acc svc)
                   nil t)))))
    (let ((trimmed (string-trim out)))
      (if (string-empty-p trimmed) nil trimmed))))

(defun my/keychain-set (service value &optional account)
  "Store VALUE in Keychain at SERVICE for ACCOUNT (default user-login-name)."
  (let ((acc (or account (user-login-name))))
    ;; Delete any prior entry first so re-saving doesn't error.
    (call-process-shell-command
     (format "security delete-generic-password -a %s -s %s 2>/dev/null"
             (shell-quote-argument acc) (shell-quote-argument service))
     nil nil)
    (call-process "security" nil nil nil
                  "add-generic-password"
                  "-a" acc "-s" service "-w" value "-A")))

(defun my/keychain-delete (service &optional account)
  "Remove SERVICE entry from Keychain for ACCOUNT."
  (call-process-shell-command
   (format "security delete-generic-password -a %s -s %s 2>/dev/null"
           (shell-quote-argument (or account (user-login-name)))
           (shell-quote-argument service))
   nil nil))
#+end_src

* Step 1.5 — Keychain migration (legacy → unified service)

#+begin_src emacs-lisp :tangle yes
(defun my/keychain-migrate-from-legacy ()
  "Copy legacy Keychain entries into `my/keychain-service' (idempotent).
Reads:
  - service `emacs-bw-master'  → service `emacs-credentials' / account `bw-master'
  - service `emacs-bw-email'   → service `emacs-credentials' / account `bw-email'
  - service `emacs-api-<KEY>'  → service `emacs-credentials' / account `<KEY>'
Doesn't overwrite values already present in the unified service.
Doesn't delete legacy entries — they remain as backup."
  (let ((moved 0))
    (cl-flet ((copy-once (legacy-svc new-acct)
                (when (and (not (my/keychain-get my/keychain-service new-acct))
                           (my/keychain-get legacy-svc))
                  (my/keychain-set my/keychain-service
                                   (my/keychain-get legacy-svc)
                                   new-acct)
                  (cl-incf moved))))
      (copy-once my/legacy-bw-keychain-service-master my/keychain-account-bw-master)
      (copy-once my/legacy-bw-keychain-service-email  my/keychain-account-bw-email)
      (dolist (env-var my/api-key-fields)
        (copy-once (concat my/legacy-api-keychain-service-prefix env-var)
                   env-var)))
    (when (> moved 0)
      (message "Bootstrap: migrated %d Keychain entries into service '%s' (legacy entries kept as backup)."
               moved my/keychain-service))))
#+end_src

* Step 2 — Bitwarden unlock

#+begin_src emacs-lisp :tangle yes
(defun my/bw-installed-p ()
  "Return t if the bw CLI is on PATH."
  (executable-find "bw"))

(defun my/bw-install ()
  "Best-effort macOS install of bw via Homebrew. No-op on other OSes."
  (when (and (eq system-type 'darwin) (executable-find "brew"))
    (start-process "install-bw" "*install-bw*" "brew" "install" "bitwarden-cli")
    (message "Installing bitwarden-cli via brew in background — restart Emacs when finished.")))

(defun my/bw-status ()
  "Return BW status keyword: 'unauthenticated', 'locked', 'unlocked', or 'unknown'."
  (let ((out (with-output-to-string
               (with-current-buffer standard-output
                 (call-process "bw" nil t nil "status")))))
    (cond
     ((string-match-p "\"unauthenticated\"" out) 'unauthenticated)
     ((string-match-p "\"locked\""          out) 'locked)
     ((string-match-p "\"unlocked\""        out) 'unlocked)
     (t 'unknown))))

(defun my/bw--unlock-internal (email master)
  "Login if needed, then unlock. Returns session token or nil.
Master is wiped from memory in the caller."
  (let* ((env (copy-sequence process-environment))
         (process-environment (cons (format "BW_PW=%s" master) env)))
    (cond
     ;; Already unlocked (rare in a fresh Emacs but possible if shell exported BW_SESSION)
     ((and (eq (my/bw-status) 'unlocked)
           (getenv "BW_SESSION"))
      (getenv "BW_SESSION"))
     ;; Need to login first
     ((eq (my/bw-status) 'unauthenticated)
      (let ((login-out (with-output-to-string
                         (with-current-buffer standard-output
                           (call-process-shell-command
                            (format "bw login %s --passwordenv BW_PW --raw 2>&1"
                                    (shell-quote-argument email))
                            nil t)))))
        (let ((tok (string-trim login-out)))
          (if (and tok (> (length tok) 30)) tok nil))))
     ;; Unlock existing login
     (t
      (let ((tok (string-trim
                  (with-output-to-string
                    (with-current-buffer standard-output
                      (call-process-shell-command
                       "bw unlock --passwordenv BW_PW --raw 2>/dev/null"
                       nil t))))))
        (if (and tok (> (length tok) 30)) tok nil))))))

(cl-defun my/bw-unlock (&optional interactive-prompt)
  "Ensure my/bw-session is set. Use cached creds from Keychain if present,
otherwise prompt for email + master and cache to Keychain on success.
INTERACTIVE-PROMPT non-nil forces a prompt even if Keychain has values.
Returns the session token or nil if the user declined / unlock failed."
  (interactive (list t))
  (when my/bw-session
    (cl-return-from my/bw-unlock my/bw-session))

  (unless (my/bw-installed-p)
    (when (and (eq system-type 'darwin)
               (y-or-n-p "bw CLI not installed. Install via brew (background)? "))
      (my/bw-install))
    (cl-return-from my/bw-unlock nil))

  ;; Migrate any legacy Keychain entries into the unified service before
  ;; reading. Idempotent and leaves the legacy entries in place as backup.
  (my/keychain-migrate-from-legacy)
  (let* ((cached-email   (and (not interactive-prompt)
                              (my/keychain-get my/keychain-service
                                               my/keychain-account-bw-email)))
         (cached-master  (and cached-email
                              (my/keychain-get my/keychain-service
                                               my/keychain-account-bw-master)))
         (use-cached     (and cached-email cached-master))
         (email          (or cached-email
                             (and (or interactive-prompt
                                      (y-or-n-p "Use Bitwarden for secrets? "))
                                  (read-string "Bitwarden email: "))))
         (master         (and email
                              (or cached-master
                                  (read-passwd "Bitwarden master password (empty to skip): ")))))
    (cond
     ;; User declined or gave empty password
     ((or (null email) (null master) (string-empty-p master))
      (message "Bitwarden skipped — falling back to macOS Keychain for secrets.")
      nil)
     (t
      (let ((session (my/bw--unlock-internal email master)))
        (cond
         (session
          (setq my/bw-session session)
          ;; Cache email + master to Keychain BEFORE wiping the local
          ;; copies. Only on a first successful unlock (when not already
          ;; cached). Master caching is opt-in via my/bw-cache-master-on-
          ;; first-launch (defaults to t).
          (unless use-cached
            (my/keychain-set my/keychain-service email my/keychain-account-bw-email)
            (when (and my/bw-cache-master-on-first-launch
                       (stringp master)
                       (not (string-empty-p master)))
              (my/keychain-set my/keychain-service master my/keychain-account-bw-master)))
          (clear-string master)
          (call-process-shell-command
           (format "bw sync --session %s >/dev/null 2>&1" (shell-quote-argument session))
           nil nil)
          (message "Bitwarden unlocked.")
          session)
         (t
          (clear-string master)
          (display-warning 'emacs-setup
                           "Bitwarden unlock failed — wrong password or 2FA required."
                           :warning)
          (when use-cached
            ;; Cached master rejected: it's stale (BW password rotated?).
            ;; Wipe so next launch re-prompts.
            (my/keychain-delete my/keychain-service my/keychain-account-bw-master))
          nil)))))))

(defun my/bw-lock ()
  "Lock the BW vault and clear the session token from memory."
  (interactive)
  (when my/bw-session
    (call-process-shell-command
     (format "bw lock --session %s >/dev/null 2>&1"
             (shell-quote-argument my/bw-session))
     nil nil)
    (setq my/bw-session nil)
    (message "Bitwarden locked.")))
#+end_src

** Caveat — master caching on first launch

The first time you enter the master password, =read-passwd= returns
the cleartext, we use it for unlock, then immediately =clear-string=
to wipe it from elisp memory. That means we can't *also* write it to
Keychain in the same call. The current implementation stashes the
email immediately and the master on the *second* successful launch
(when the user re-enters and we cache before clearing).

If you want master cached on first launch instead, set
=my/bw-cache-master-on-first-launch= to t and accept that the master
sits in Emacs memory for the ~20 ms between unlock and Keychain write.

#+begin_src emacs-lisp :tangle yes
(defvar my/bw-cache-master-on-first-launch t
  "When t, cache master to Keychain on first successful unlock
(slight extra exposure window). When nil, second unlock caches.")
#+end_src

#+begin_src emacs-lisp :tangle yes
(require 'cl-lib)
(require 'json)

(defun my/bw-get-item (item-name)
  "Return parsed alist for ITEM-NAME, or nil if missing / no session."
  (when my/bw-session
    (let* ((out (with-output-to-string
                  (with-current-buffer standard-output
                    (call-process-shell-command
                     (format "bw get item %s --session %s 2>/dev/null"
                             (shell-quote-argument item-name)
                             (shell-quote-argument my/bw-session))
                     nil t))))
           (direct (when (and out (string-prefix-p "{" (string-trim out)))
                     (condition-case nil
                         (json-parse-string out
                                            :object-type 'alist
                                            :null-object nil
                                            :false-object nil)
                       (error nil)))))
      (or direct
          ;; Duplicate item names make `bw get item NAME' ambiguous. Fall back
          ;; to exact-name search and choose the candidate with most fields.
          (let* ((list-out (with-output-to-string
                             (with-current-buffer standard-output
                               (call-process-shell-command
                                (format "bw list items --search %s --session %s 2>/dev/null"
                                        (shell-quote-argument item-name)
                                        (shell-quote-argument my/bw-session))
                                nil t))))
                 (items (when (and list-out
                                   (string-prefix-p "[" (string-trim list-out)))
                          (condition-case nil
                              (json-parse-string list-out
                                                 :object-type 'alist
                                                 :array-type 'list
                                                 :null-object nil
                                                 :false-object nil)
                            (error nil))))
                 (exact (cl-remove-if-not
                         (lambda (item)
                           (string= (cdr (assoc 'name item)) item-name))
                         (or items '()))))
            (car
             (sort exact
                   (lambda (a b)
                     (> (length (or (cdr (assoc 'fields a)) []))
                        (length (or (cdr (assoc 'fields b)) [])))))))))))

(defun my/bw-item-password (item)
  "Extract login.password from a parsed BW item."
  (when item
    (let ((login (cdr (assoc 'login item))))
      (and login (cdr (assoc 'password login))))))

(defun my/bw-item-username (item)
  "Extract login.username from a parsed BW item."
  (when item
    (let ((login (cdr (assoc 'login item))))
      (and login (cdr (assoc 'username login))))))

(defun my/bw-item-fields-list (item)
  "Return ITEM custom fields as a list, tolerating vector or list JSON shapes."
  (let ((fields (and item (cdr (assoc 'fields item)))))
    (cond
     ((vectorp fields) (append fields nil))
     ((listp fields) fields)
     (t nil))))

(defun my/bw-item-field (item field-name)
  "Extract custom field FIELD-NAME from a parsed BW item."
  (when item
    (let ((fields (my/bw-item-fields-list item)))
      (cl-loop for f in fields
               when (string= (cdr (assoc 'name f)) field-name)
               return (cdr (assoc 'value f))))))

(defun my/bw--encode-json (json)
  "Return Bitwarden CLI's encoded representation of JSON, or error."
  (with-temp-buffer
    (insert json)
    (let ((status (call-process-region (point-min) (point-max)
                                       "bw" t t nil "encode")))
      (unless (zerop status)
        (error "BW encode failed: %s" (string-trim (buffer-string)))))
    (string-trim (buffer-string))))

(defun my/bw--call (&rest args)
  "Run `bw' with ARGS, returning trimmed stdout or raising an error."
  (with-temp-buffer
    (let ((status (apply #'call-process "bw" nil t nil args))
          (cmd (mapconcat #'identity (cl-subseq args 0 (min 2 (length args))) " ")))
      (unless (zerop status)
        (error "BW command failed: bw %s (details redacted)\n%s"
               cmd
               (string-trim (buffer-string)))))
    (string-trim (buffer-string))))

(defun my/bw-create-item (name &rest plist)
  "Create a BW login item NAME.
PLIST: :username, :password, :fields (alist of (name . value))."
  (unless my/bw-session
    (error "BW not unlocked; cannot create item %s" name))
  (let* ((fields (cl-loop for (n . v) in (plist-get plist :fields)
                          collect `((name . ,n) (value . ,v) (type . 0))))
         (payload `((type . 1)
                    (name . ,name)
                    (login . ((username . ,(or (plist-get plist :username) ""))
                              (password . ,(or (plist-get plist :password) ""))))
                    (fields . ,(vconcat fields))))
         (json (json-serialize payload))
         (encoded (my/bw--encode-json json)))
    (my/bw--call "create" "item" encoded "--session" my/bw-session)))

(defun my/bw-update-item (item-name &rest plist)
  "Update existing BW item ITEM-NAME with new field values.
PLIST: same keys as my/bw-create-item. Only the keys present are touched."
  (let ((item (my/bw-get-item item-name)))
    (unless item (error "BW item %s not found" item-name))
    (let* ((id      (cdr (assoc 'id item)))
           (login   (or (cdr (assoc 'login item)) '()))
           (fields  (my/bw-item-fields-list item))
           (new-user (or (plist-get plist :username)
                         (cdr (assoc 'username login))
                         ""))
           (new-pw   (or (plist-get plist :password)
                         (cdr (assoc 'password login))
                         ""))
           (new-flds (cl-reduce
                      (lambda (acc nv)
                        (let* ((n (car nv)) (v (cdr nv))
                               (other (cl-remove-if
                                       (lambda (f) (string= (cdr (assoc 'name f)) n))
                                       acc)))
                          (append other (list `((name . ,n) (value . ,v) (type . 0))))))
                      (plist-get plist :fields)
                      :initial-value fields))
           (payload `((id . ,id)
                      (type . 1)
                      (name . ,item-name)
                      (login . ((username . ,new-user) (password . ,new-pw)))
                      (fields . ,(vconcat new-flds))))
           (encoded (my/bw--encode-json (json-serialize payload))))
      (my/bw--call "edit" "item" id encoded "--session" my/bw-session))))

* Step 3.5 — Credentials migration (legacy BW items → unified item)

#+begin_src emacs-lisp :tangle yes
(defun my/-blank-p (s) (or (null s) (and (stringp s) (string-empty-p s))))

(defun my/bw-migrate-from-legacy-items ()
  "Consolidate legacy BW items into `my/bw-credentials-item' (idempotent).
Copies non-blank fields from `my/bw-legacy-github-item' (GitHub bundle)
and `my/bw-legacy-api-item' (API keys) into the unified item. Never
overwrites a non-blank field already in the unified item. Leaves
legacy items in place as backup. No-op when BW is locked."
  (when my/bw-session
    (let* ((dest (my/bw-get-item my/bw-credentials-item))
           (gh   (my/bw-get-item my/bw-legacy-github-item))
           (api  (my/bw-get-item my/bw-legacy-api-item)))
      (when (or gh api)
        (let* ((dest-user (and dest (my/bw-item-username dest)))
               (dest-pw   (and dest (my/bw-item-password dest)))
               (gh-user   (and gh   (my/bw-item-username gh)))
               (gh-pw     (and gh   (my/bw-item-password gh)))
               (new-user  (if (my/-blank-p dest-user) (or gh-user "") dest-user))
               (new-pw    (if (my/-blank-p dest-pw)   (or gh-pw "")   dest-pw))
               (gh-flds   '("Email" "Fullname" "Repo" "GitCryptKey"))
               (to-add    nil)
               (added     nil))
          (dolist (f gh-flds)
            (let ((cur (and dest (my/bw-item-field dest f)))
                  (src (and gh   (my/bw-item-field gh f))))
              (when (and (my/-blank-p cur) (not (my/-blank-p src)))
                (push (cons f src) to-add)
                (push f added))))
          (dolist (env-var my/api-key-fields)
            (let ((cur (and dest (my/bw-item-field dest env-var)))
                  (src (and api  (my/bw-item-field api env-var))))
              (when (and (my/-blank-p cur) (not (my/-blank-p src)))
                (push (cons env-var src) to-add)
                (push env-var added))))
          (let ((promoted-user (and (my/-blank-p dest-user) (not (my/-blank-p gh-user))))
                (promoted-pw   (and (my/-blank-p dest-pw)   (not (my/-blank-p gh-pw)))))
            (cond
             ((null dest)
              (my/bw-create-item my/bw-credentials-item
                                 :username new-user
                                 :password new-pw
                                 :fields   (nreverse to-add))
              (message "Bootstrap: created BW item '%s' (migrated %d field(s)%s%s from %s)."
                       my/bw-credentials-item
                       (length to-add)
                       (if promoted-user " + username" "")
                       (if promoted-pw   " + token" "")
                       (cond ((and gh api) (format "'%s' + '%s'" my/bw-legacy-github-item my/bw-legacy-api-item))
                             (gh           (format "'%s'" my/bw-legacy-github-item))
                             (api          (format "'%s'" my/bw-legacy-api-item))
                             (t            "<none>"))))
             ((or to-add promoted-user promoted-pw)
              (my/bw-update-item my/bw-credentials-item
                                 :username new-user
                                 :password new-pw
                                 :fields   (nreverse to-add))
              (message "Bootstrap: enriched BW item '%s' from legacy items (added: %s)."
                       my/bw-credentials-item
                       (mapconcat #'identity (nreverse added) ", "))))))))))

* Step 4 — GitHub identity bundle (token + email + fullname + repo)

The =emacs-credentials= BW item carries everything we need to clone
the user's private data repo (migration from the older =github-cli-token=
item runs once on first launch — see Step 3.5):
- =login.username=  → GitHub handle (e.g. "alice")
- =login.password=  → PAT (=ghp_...= classic or =github_pat_...= fine-grained)
- custom =Email=    → git commit email
- custom =Fullname= → git commit name
- custom =Repo=     → private repo name (e.g. "emacs-data", not "alice/emacs-data")

#+begin_src emacs-lisp :tangle yes
(defun my/github-token-valid-p (token)
  "Non-nil unless GitHub authoritatively rejects TOKEN with HTTP 401.
Network failures, 5xx, 403 (rate-limit), and any other status preserve
the cached token so a fresh PAT prompt only fires when GitHub actually
says 'Bad credentials'. Offline / outage launches keep using the
cached token."
  (when (and token (not (string-empty-p token)))
    (let ((code (string-trim
                 (with-output-to-string
                   (with-current-buffer standard-output
                     (call-process-shell-command
                      (format "curl -sS --max-time 5 -o /dev/null -w %%{http_code} -H %s https://api.github.com/user 2>/dev/null"
                              (shell-quote-argument
                               (format "Authorization: Bearer %s" token)))
                      nil t))))))
      (not (string= code "401")))))

(defun my/github-prompt-identity ()

(cl-defun my/github-load-identity ()
  "Read GitHub bundle from BW item; fill in any missing fields (and
update BW). Returns plist with :username :token :email :fullname :repo,
or nil if BW unavailable / user declined.

Prompts only for fields that are missing or invalid:
  - token missing or `gh api user' returns 401 → re-prompt token
  - any of username / email / fullname / repo missing → prompt that one"
  (unless my/bw-session
    (cl-return-from my/github-load-identity nil))
  (let* ((item     (my/bw-get-item my/bw-credentials-item))
         (username (and item (my/bw-item-username item)))
         (token    (and item (my/bw-item-password item)))
         (email    (and item (my/bw-item-field item "Email")))
         (fullname (and item (my/bw-item-field item "Fullname")))
         (repo     (and item (my/bw-item-field item "Repo")))
         ;; Read but DON'T require — git-crypt is opt-in per repo.
         ;; The git-crypt step will prompt if the repo turns out to need it.
         (git-crypt-key (and item (my/bw-item-field item "GitCryptKey")))
         (need-token (or (my/-blank-p token)
                         (not (my/github-token-valid-p token))))
         (need-username (my/-blank-p username))
         (need-email    (my/-blank-p email))
         (need-fullname (my/-blank-p fullname))
         (need-repo     (my/-blank-p repo))
         (any-missing (or need-token need-username need-email need-fullname need-repo)))
    (cond
     ;; Item present and complete — nothing to ask.
     ((and item (not any-missing))
      (message "Bootstrap: BW '%s' complete (username=%s, repo=%s)."
               my/bw-credentials-item username repo)
      (list :username username :token token :email email :fullname fullname
            :repo repo :git-crypt-key git-crypt-key))
     ;; Item missing entirely — ask upfront whether to create.
     ((and (not item)
           (not (y-or-n-p (format "BW item '%s' not found. Create it now? "
                                  my/bw-credentials-item))))
      (message "Bootstrap: skipped creating BW '%s'; gh / repo-sync disabled this session."
               my/bw-credentials-item)
      nil)
     ;; Either item missing (and user said yes), or item present with gaps.
     (t
      (when need-token
        (display-warning 'emacs-setup
                         (if item
                             "GitHub token in BW missing or invalid — re-prompting."
                           (format "Creating BW item '%s'." my/bw-credentials-item))
                         :warning))
      (let* ((new-username (if need-username (read-string "GitHub login (handle): ")    username))
             (new-token    (if need-token    (read-passwd "GitHub personal access token (repo scope): ") token))
             (new-email    (if need-email    (read-string "Git commit email: ")         email))
             (new-fullname (if need-fullname (read-string "Git commit name: ")          fullname))
             (new-repo     (if need-repo     (read-string "Private data repo name (e.g. emacs-data): ") repo))
             (fields       `(("Email"    . ,new-email)
                             ("Fullname" . ,new-fullname)
                             ("Repo"     . ,new-repo))))
        (if item
            (progn
              (my/bw-update-item my/bw-credentials-item
                                 :username new-username
                                 :password new-token
                                 :fields   fields)
              (message "Bootstrap: BW '%s' updated (filled %s)."
                       my/bw-credentials-item
                       (mapconcat #'symbol-name
                                  (delq nil (list (and need-token    :token)
                                                  (and need-username :username)
                                                  (and need-email    :email)
                                                  (and need-fullname :fullname)
                                                  (and need-repo     :repo)))
                                  ", ")))
          (my/bw-create-item my/bw-credentials-item
                             :username new-username
                             :password new-token
                             :fields   fields)
          (message "Bootstrap: BW '%s' created." my/bw-credentials-item))
        (list :username new-username :token new-token
              :email new-email :fullname new-fullname :repo new-repo
              :git-crypt-key git-crypt-key))))))

(defun my/-repo-copy-local-only-files (local-dir repo-dir)
  "Copy files that exist only in LOCAL-DIR into REPO-DIR.
For data files present on both sides with different content, keep the repo
version and record the relative path as a conflict.  For files under config/,
keep the freshly seeded local version because those scripts are required to
finish the current bootstrap load.  Return a plist with :imported,
:config-updated and :conflicts lists."
  (let ((imported nil)
        (config-updated nil)
        (conflicts nil))
    (dolist (src (directory-files-recursively local-dir ".*" t))
      (when (file-regular-p src)
        (let* ((rel (file-relative-name src local-dir))
               (dst (expand-file-name rel repo-dir)))
          (unless (or (string-prefix-p ".git/" rel)
                      (member (file-name-nondirectory rel)
                              '(".DS_Store" ".localized")))
            (cond
             ((not (file-exists-p dst))
              (make-directory (file-name-directory dst) t)
              (copy-file src dst t t t t)
              (push rel imported))
             ((and (string-prefix-p "config/" rel)
                   (not (my/-file-same-content-p src dst)))
              (copy-file src dst t t t t)
              (push rel config-updated))
             ((not (my/-file-same-content-p src dst))
              (push rel conflicts)))))))
    (list :imported (nreverse imported)
          :config-updated (nreverse config-updated)
          :conflicts (nreverse conflicts))))

(defun my/-repo-adopt-existing-dir (identity current-remote)
  "Adopt existing my/data-dir into the configured GitHub repo.
The GitHub repo is authoritative for paths that exist on both sides.
Local-only files are imported and pushed.  The original directory is
always kept as a timestamped backup so conflicting local versions can
be recovered manually."
  (let* ((user (plist-get identity :username))
         (repo (plist-get identity :repo))
         (dir my/data-dir)
         (dir-target (directory-file-name dir))
         (dir-parent (file-name-directory dir-target))
         (stamp (format-time-string "%Y%m%d-%H%M%S"))
         (backup (format "%s.bak-%s" dir-target stamp))
         (tmp (make-temp-file (expand-file-name ".emacs-data-adopt-" dir-parent) t)))
    (if (not (my/-repo-exists-on-gh-p user repo))
        (progn
          (delete-directory tmp t)
          (display-warning
           'emacs-setup
           (format "%s exists with content, but %s/%s does not exist on GitHub.
Leaving local files untouched. Create the repo first or move %s aside manually."
                   dir user repo dir)
           :warning))
      (message "Bootstrap: cloning %s/%s into temporary adoption dir %s…" user repo tmp)
      (unless (zerop (call-process "gh" nil nil nil "repo" "clone"
                                   (format "%s/%s" user repo) tmp))
        (delete-directory tmp t)
        (error "Could not clone %s/%s for adoption" user repo))
      (let* ((result (my/-repo-copy-local-only-files dir tmp))
             (imported (plist-get result :imported))
             (config-updated (plist-get result :config-updated))
             (conflicts (plist-get result :conflicts)))
        (message "Bootstrap: moving existing %s → %s before adopting clone" dir backup)
        (condition-case err
            (progn
              (rename-file dir-target backup)
              (my/-repo-write-adoption-report backup imported config-updated conflicts user repo)
              (rename-file tmp dir-target)
              (when (or imported config-updated)
                (my/-repo-commit-and-push-if-needed
                 dir "sync: import bootstrap adoption files"))
              (message
               "Bootstrap: adopted %s/%s into %s. Imported %d local-only file(s); updated %d config file(s); kept %d remote data conflict(s). Previous local tree: %s"
               user repo dir (length imported) (length config-updated) (length conflicts) backup))
          (error
           (when (and (not (file-exists-p dir)) (file-exists-p backup))
             (rename-file backup dir-target))
           (when (file-exists-p tmp)
             (delete-directory tmp t))
           (display-warning
            'emacs-setup
            (format "Repo adoption failed and was rolled back: %s"
                    (error-message-string err))
            :warning)))))))

(defun my/repo-sync (identity)
  "Sync the user's private data repo into my/data-dir.
IDENTITY is the plist returned by my/github-load-identity."
  (let* ((user (plist-get identity :username))
         (repo (plist-get identity :repo))
         (expected-url (format "https://github.com/%s/%s.git" user repo))
         (dir my/data-dir))
    (cond
     ;; Existing clone with matching remote → pull
     ((and (file-directory-p dir)
           (let ((url (my/git-remote-url dir)))
             (and url (or (string= url expected-url)
                          (string= url (replace-regexp-in-string "\\.git\\'" "" expected-url))))))
      (message "Bootstrap: my/data-dir is a clone of %s/%s — pulling…" user repo)
      (let ((default-directory dir))
        ;; --rebase + --autostash: replay local commits on top of fetched
        ;; remote; stash any unstaged changes for the duration. Conflicts
        ;; cause the rebase to pause; we detect that and abort below.
        (call-process-shell-command
         "git pull --rebase --autostash origin main 2>&1"
         nil "*repo-pull*" nil)
        (with-current-buffer (get-buffer-create "*repo-pull*")
          (when (string-match-p "CONFLICT\\|Could not apply" (buffer-string))
            (display-warning 'emacs-setup
                             (concat "Repo pull conflict in " dir
                                     " — leaving repo alone. Resolve manually then restart Emacs.")
                             :warning)))))
     ;; Empty / absent → clone or create
     ((or (not (file-exists-p dir))
          (and (file-directory-p dir)
               (null (directory-files dir nil "\\`[^.]"))))
      (my/-repo-fetch-or-create identity))
     ;; Has content but wrong/no remote — adopt local-only files into the
     ;; GitHub clone while keeping the remote authoritative for conflicts.
     (t
      (let ((current-remote (or (my/git-remote-url dir) "(no git remote)")))
        (cond
         ((y-or-n-p
           (format "%s exists with content (remote: %s) and is NOT %s/%s.
Adopt it into the GitHub repo? Remote wins data conflicts; seeded config/ scripts win so startup can finish; local-only files are imported; full local backup is kept. "
                   dir current-remote user repo))
          (my/-repo-adopt-existing-dir identity current-remote))
         (t
          (display-warning 'emacs-setup
                           (format "%s exists with content but is not a clone of %s/%s — leaving alone.
To adopt later, M-x my/repo-sync (will re-prompt) or move %s manually."
                                   dir user repo dir)
                           :warning))))))))

* Step 7.5 — git-crypt unlock for the private repo

If the user's private data repo uses git-crypt to encrypt sensitive
files (typical for org files that contain personal data, API keys
they've inlined, GTD content, etc.), we need to provide the symmetric
key on each new Mac. The key is stored as the =GitCryptKey= custom
field of =my/bw-credentials-item= (the same BW item that holds the GitHub
PAT + identity bundle), base64-encoded so it round-trips cleanly
through BW's string fields.

Three cases handled:

- *Repo uses git-crypt + key in BW*  → install git-crypt if missing;
  unlock immediately.
- *Repo uses git-crypt + key NOT in BW* (new Mac, never set up here)
  → prompt user to paste base64-encoded key from another Mac; save
  to BW for next time; unlock.
- *Repo doesn't use git-crypt* (no =.gitattributes= filter=git-crypt
  entries) → skip entirely.

To export the key on a Mac that already has it set up, run
=git-crypt export-key /tmp/k && base64 -i /tmp/k= and copy that
string when prompted on the new Mac.

#+begin_src emacs-lisp :tangle yes
(defvar my/git-crypt-keychain-service "emacs-git-crypt-key"
  "macOS Keychain service for the cached git-crypt key (base64-encoded).
Used as a fallback when Bitwarden isn't available — same pattern as
the API-key Keychain fallback. The key is stored under (account=$USER,
service=this) and contains the base64 string verbatim.")

(defun my/git-crypt-installed-p () (executable-find "git-crypt"))

(defun my/git-crypt-install ()
  "Install git-crypt on macOS via brew. Returns t on success."
  (when (and (eq system-type 'darwin) (executable-find "brew"))
    (call-process-shell-command "brew install git-crypt" nil "*install-git-crypt*" t)
    (executable-find "git-crypt")))

(defun my/repo-uses-git-crypt-p (dir)
  "Return t if DIR has a .gitattributes entry that pipes through git-crypt."
  (let ((ga (expand-file-name ".gitattributes" dir)))
    (and (file-exists-p ga)
         (with-temp-buffer
           (insert-file-contents ga)
           (re-search-forward "filter=git-crypt" nil t)))))

(defun my/repo-already-unlocked-p (dir)
  "Cheap check: a key file is materialised under .git/git-crypt/keys/."
  (file-exists-p (expand-file-name ".git/git-crypt/keys/default" dir)))

(defun my/-normalize-base64 (s)
  "Strip whitespace and translate base64url chars to standard base64.
`pbcopy' / shell pipelines often produce trailing newlines; some tools
emit `_' / `-' instead of `/' / `+'."
  (when s
    (replace-regexp-in-string
     "_" "/"
     (replace-regexp-in-string
      "-" "+"
      (replace-regexp-in-string "[ \t\n\r]" "" s)))))

(defun my/-valid-base64-p (s)
  "Return non-nil if S (after normalization) decodes as base64."
  (and s
       (condition-case nil
           (progn (base64-decode-string (my/-normalize-base64 s)) t)
         (error nil))))

(defun my/git-crypt-key-prompt-and-save ()
  "Prompt for a base64-encoded git-crypt key. Validate before saving.
On success, save the NORMALIZED form to whichever secret store is
available:
  - Bitwarden (preferred) — custom field GitCryptKey on my/bw-credentials-item
  - macOS Keychain (fallback when BW isn't unlocked) —
    service=my/git-crypt-keychain-service
Returns the normalized key or nil.

Tip: on a Mac that already has the repo unlocked, generate the input
with:  `git-crypt export-key /tmp/k && base64 -i /tmp/k | pbcopy`"
  (let* ((entered (read-string
                   "Base64-encoded git-crypt key (paste, empty to skip): "))
         (normalized (and entered (my/-normalize-base64 entered))))
    (cond
     ((my/-blank-p entered)
      (display-warning
       'emacs-setup
       "git-crypt key not provided — encrypted files in the private repo will appear as binary on this Mac."
       :warning)
      nil)
     ((not (my/-valid-base64-p entered))
      (display-warning
       'emacs-setup
       "What you pasted isn't valid base64. Re-export with:
  git-crypt export-key /tmp/k && base64 -i /tmp/k | pbcopy
then paste again (M-x my/git-crypt-step). NOT saved."
       :warning)
      nil)
     (t
      (cond
       (my/bw-session
        (my/bw-update-item my/bw-credentials-item
                           :fields `(("GitCryptKey" . ,normalized)))
        (message "Bootstrap: git-crypt key saved to BW '%s' (GitCryptKey field)."
                 my/bw-credentials-item))
       (t
        (my/keychain-set my/git-crypt-keychain-service normalized)
        (message "Bootstrap: git-crypt key saved to macOS Keychain (service %s)."
                 my/git-crypt-keychain-service)))
      normalized))))

(defun my/git-crypt-key-resolve (identity)
  "Return the base64-encoded git-crypt key from the best available source.
Order:
  1. IDENTITY plist's :git-crypt-key  (= BW's GitCryptKey field, loaded
     earlier by my/github-load-identity when BW was unlocked)
  2. macOS Keychain (service my/git-crypt-keychain-service)
  3. Prompt the user and persist (to BW if unlocked, else Keychain)

Returns nil if the user declined to provide a key."
  (or
   ;; 1. From BW (already in identity)
   (let ((v (plist-get identity :git-crypt-key)))
     (and (not (my/-blank-p v))
          (progn (message "Bootstrap: git-crypt key from BW '%s' (GitCryptKey)."
                          my/bw-credentials-item)
                 v)))
   ;; 2. macOS Keychain fallback
   (let ((v (my/keychain-get my/git-crypt-keychain-service)))
     (and (not (my/-blank-p v))
          (progn (message "Bootstrap: git-crypt key from macOS Keychain (service %s)."
                          my/git-crypt-keychain-service)
                 v)))
   ;; 3. Prompt + persist
   (my/git-crypt-key-prompt-and-save)))

(defun my/git-crypt-clear-key ()
  "Clear the cached git-crypt key from BOTH stores (BW + macOS Keychain).
Use when you've saved an invalid key and want bootstrap to re-prompt."
  (interactive)
  (let ((cleared-anywhere nil))
    ;; BW (if unlocked or unlockable)
    (unless my/bw-session (my/bw-unlock t))
    (when my/bw-session
      (my/bw-update-item my/bw-credentials-item :fields '(("GitCryptKey" . "")))
      (message "Cleared GitCryptKey in BW '%s'." my/bw-credentials-item)
      (setq cleared-anywhere t))
    ;; macOS Keychain (always; harmless if absent)
    (my/keychain-delete my/git-crypt-keychain-service)
    (message "Cleared Keychain entry %s." my/git-crypt-keychain-service)
    (setq cleared-anywhere t)
    (when cleared-anywhere
      (message "git-crypt key cleared. Re-run M-x my/git-crypt-step to enter a new one."))))

(defun my/git-crypt-unlock (dir base64-key)
  "Unlock the git-crypt repo at DIR using BASE64-KEY (a string).
Writes the decoded key to a temp file, runs `git-crypt unlock`, then
deletes the temp file regardless of outcome.

If the key fails to decode OR git-crypt rejects it, AUTO-WIPE the bad
value from BW (so next launch re-prompts) and surface a clear warning.
Bootstrap continues — it does NOT halt."
  (let ((keyfile (make-temp-file "git-crypt-key-"))
        (ok nil))
    (unwind-protect
        (condition-case err
            (progn
              (with-temp-file keyfile
                (set-buffer-multibyte nil)
                (insert (base64-decode-string (my/-normalize-base64 base64-key))))
              (let ((default-directory dir))
                (call-process-shell-command
                 (format "git-crypt unlock %s 2>&1" (shell-quote-argument keyfile))
                 nil "*git-crypt-unlock*" nil))
              (with-current-buffer (get-buffer-create "*git-crypt-unlock*")
                (cond
                 ((string-match-p "Error\\|fatal\\|invalid" (buffer-string))
                  (display-warning
                   'emacs-setup
                   "git-crypt unlock failed — see *git-crypt-unlock* buffer. Clearing cached git-crypt key from BW + Keychain so next bootstrap re-prompts."
                   :warning)
                  (when my/bw-session
                    (my/bw-update-item my/bw-credentials-item
                                       :fields '(("GitCryptKey" . ""))))
                  (my/keychain-delete my/git-crypt-keychain-service))
                 (t
                  (setq ok t)
                  (message "Bootstrap: git-crypt unlocked %s." dir)))))
          (error
           ;; base64-decode-string failure or write-file failure.
           (display-warning
            'emacs-setup
            (format "git-crypt unlock errored at decode step: %s.
Clearing cached git-crypt key from BW + Keychain so next bootstrap re-prompts."
                    (error-message-string err))
            :warning)
           (when my/bw-session
             (my/bw-update-item my/bw-credentials-item
                                :fields '(("GitCryptKey" . ""))))
           (my/keychain-delete my/git-crypt-keychain-service)))
      (delete-file keyfile))
    ok))

(defun my/git-crypt-step (identity)
  "Top-level dispatcher invoked from my/bootstrap after repo sync.
IDENTITY is the plist returned by my/github-load-identity (may contain
a :git-crypt-key already, in which case no prompt is needed).
No-op if the repo doesn't use git-crypt or is already unlocked."
  (cond
   ((not (and (file-directory-p my/data-dir)
              (my/repo-uses-git-crypt-p my/data-dir)))
    (message "Bootstrap: repo doesn't use git-crypt; skipping unlock."))
   ((my/repo-already-unlocked-p my/data-dir)
    (message "Bootstrap: git-crypt already unlocked on this Mac."))
   (t
    (unless (my/git-crypt-installed-p)
      (message "Bootstrap: installing git-crypt via brew…")
      (my/git-crypt-install))
    (cond
     ((not (my/git-crypt-installed-p))
      (display-warning
       'emacs-setup
       "git-crypt could not be installed — install manually with: brew install git-crypt"
       :warning))
     (t
      ;; Resolver: BW → Keychain → prompt + persist (to whichever is available).
      (let ((key (my/git-crypt-key-resolve identity)))
        (when key
          (my/git-crypt-unlock my/data-dir key))))))))

* Step 8 — API keys: load from one BW item (preferred) or Keychain (fallback)

#+begin_src emacs-lisp :tangle yes
(defun my/bw-api-key-set (env-var value)
  "Persist ENV-VAR API key VALUE in the consolidated Bitwarden API item."
  (unless my/bw-session
    (error "BW not unlocked; cannot save %s" env-var))
  (let ((fields `((,env-var . ,value))))
    (if (my/bw-get-item my/bw-credentials-item)
        (my/bw-update-item my/bw-credentials-item :fields fields)
      (my/bw-create-item my/bw-credentials-item
                         :username "emacs"
                         :password ""
                         :fields fields)))
  (let* ((item (my/bw-get-item my/bw-credentials-item))
         (saved (and item (my/bw-item-field item env-var))))
    (unless (and saved (string= saved value))
      (error "BW verification failed: %s was not saved to item %s"
             env-var my/bw-credentials-item))))

(defun my/api-key--opted-out-p (value)
  "Return t when VALUE is the skip sentinel."
  (and (stringp value) (string= value my/api-skipped-sentinel)))

(defun my/api-key--persist-sentinel (env-var)
  "Persist the skip sentinel for ENV-VAR so we don't re-prompt next launch."
  (cond
   (my/bw-session
    (my/bw-api-key-set env-var my/api-skipped-sentinel))
   (t
    (my/keychain-set my/keychain-service my/api-skipped-sentinel env-var))))

(defun my/api-key-reset (env-var)
  "Clear any cached value or skip sentinel for ENV-VAR so the next
bootstrap prompts again. Deletes the BW custom field (if BW is
unlocked) AND the Keychain account, so the user can recover from
either backend independently. Run as `M-x my/api-key-reset'."
  (interactive
   (list (completing-read "Reset API key (env var): "
                          my/api-key-fields nil t)))
  (when my/bw-session
    (when (my/bw-get-item my/bw-credentials-item)
      (my/bw-update-item my/bw-credentials-item
                         :fields `((,env-var . "")))))
  (my/keychain-delete my/keychain-service env-var)
  (setenv env-var nil)
  (message "Reset %s — next bootstrap will prompt again." env-var))

(defun my/api-keys-load ()
  "Populate provider API key env vars.
Resolution order, per `my/api-key-fields' entry:
  1. `my/bw-credentials-item' custom field (when BW is unlocked).
  2. Legacy per-provider BW items (migrated into the consolidated item).
  3. `my/keychain-service' / account = env-var (when BW is unavailable).
  4. Prompt + persist (BW if unlocked, else Keychain).

A field whose stored value equals `my/api-skipped-sentinel' is treated
as 'user opted out': no setenv, no prompt. Use `M-x my/api-key-reset'
to clear that sentinel for a single key."
  (let ((api-item (and my/bw-session (my/bw-get-item my/bw-credentials-item)))
        loaded prompted opted-out skipped)
    (dolist (env-var my/api-key-fields)
      (let* ((value nil)
             (source nil)
             (opt-out nil))
        ;; 1. Consolidated BW item custom field
        (when (and api-item (my/-blank-p value))
          (let ((v (my/bw-item-field api-item env-var)))
            (cond
             ((my/api-key--opted-out-p v)
              (setq opt-out t source 'bw))
             ((not (my/-blank-p v))
              (setq value v source 'bw)))))
        ;; 2. Legacy per-provider BW items -> consolidated item migration.
        (when (and my/bw-session (my/-blank-p value) (not opt-out))
          (let* ((legacy-name (cdr (assoc env-var my/bw-legacy-api-items)))
                 (legacy-item (and legacy-name (my/bw-get-item legacy-name)))
                 (v (and legacy-item
                         (or (my/bw-item-password legacy-item)
                             (my/bw-item-field legacy-item env-var)
                             (my/bw-item-field legacy-item "Key")
                             (my/bw-item-field legacy-item "API_KEY")))))
            (unless (my/-blank-p v)
              (my/bw-api-key-set env-var v)
              (setq api-item (my/bw-get-item my/bw-credentials-item))
              (setq value v source 'bw-legacy))))
        ;; 3. Keychain fallback only when Bitwarden is unavailable.
        (when (and (not my/bw-session) (my/-blank-p value) (not opt-out))
          (let ((v (my/keychain-get my/keychain-service env-var)))
            (cond
             ((my/api-key--opted-out-p v)
              (setq opt-out t source 'keychain))
             ((not (my/-blank-p v))
              (setq value v source 'keychain)))))
        ;; 4. Prompt + persist (only when nothing found and not opted out).
        (when (and (my/-blank-p value) (not opt-out))
          (if noninteractive
              (push env-var skipped)
            (let ((entered (read-passwd
                            (format "%s (empty = skip permanently, won't ask again): "
                                    env-var))))
              (cond
               ((my/-blank-p entered)
                ;; Empty input = explicit opt-out. Persist the sentinel so
                ;; future launches don't re-prompt.
                (my/api-key--persist-sentinel env-var)
                (setq opt-out t source 'prompt)
                (push env-var opted-out))
               (t
                (setq value entered source 'prompt)
                (push env-var prompted)
                (if my/bw-session
                    (progn
                      (my/bw-api-key-set env-var entered)
                      (setq api-item (my/bw-get-item my/bw-credentials-item)))
                  (my/keychain-set my/keychain-service entered env-var)))))))
        (when (and value (not (my/-blank-p value)))
          (setenv env-var value)
          (push (format "%s(%s)" env-var (symbol-name source)) loaded))))
    ;; Visible progress summary
    (when loaded
      (message "Bootstrap: API keys loaded: %s" (mapconcat #'identity (nreverse loaded) ", ")))
    (when prompted
      (message "Bootstrap: API keys created via prompt + stored to %s: %s"
               (if my/bw-session
                   (format "Bitwarden item '%s'" my/bw-credentials-item)
                 (format "Keychain service '%s'" my/keychain-service))
               (mapconcat #'identity (nreverse prompted) ", ")))
    (when opted-out
      (message "Bootstrap: opted out (will NOT re-prompt; M-x my/api-key-reset to revive): %s"
               (mapconcat #'identity (nreverse opted-out) ", ")))
    (when skipped
      (message "Bootstrap: skipped (noninteractive, no value found): %s"
               (mapconcat #'identity (nreverse skipped) ", ")))))
#+end_src

* Step 9 — Orchestration (runs synchronously, in order)

#+begin_src emacs-lisp :tangle yes
(defun my/bootstrap ()
  "Run the full first-launch bootstrap: BW unlock, GitHub identity,
gh install/auth, repo sync, starter generation, API key loading.
Idempotent; safe to invoke as M-x my/bootstrap to re-run."
  (interactive)
  (message "Bootstrap: starting…")

  ;; 1-2. BW
  (my/bw-unlock)
  (if my/bw-session
      (message "Bootstrap: Bitwarden vault unlocked.")
    (message "Bootstrap: Bitwarden not in use (skipped or unavailable); will use macOS Keychain for secrets."))

  ;; 2b. Consolidate legacy BW items (github-cli-token, emacs-api-keys) into
  ;; my/bw-credentials-item. No-op when BW is locked or nothing to migrate.
  (my/bw-migrate-from-legacy-items)

  ;; 3-4. GitHub identity
  (let ((id (my/github-load-identity)))
    ;; 4b. Global git identity — fills `git config --global user.{name,email}'
    ;; from the BW bundle on Macs that don't have it set yet. Idempotent.
    (my/git-identity-ensure-global id)
    (cond
     ((and id (plist-get id :token))
      ;; 5. gh install + auth
      (cond
       ((my/gh-installed-p)
        (message "Bootstrap: gh CLI present."))
       (t
        (message "Bootstrap: gh CLI missing — installing via brew (background)…")
        (my/gh-install)))
      (when (my/gh-installed-p)
        (cond
         ((my/gh-authenticated-p)
          (message "Bootstrap: gh already authenticated."))
         (t
          (message "Bootstrap: authenticating gh with BW-stored token…")
          (my/gh-login-with-token (plist-get id :token))))
        ;; 6. Repo sync
        (cond
         ((and (plist-get id :username) (plist-get id :repo))
          (my/repo-sync id)
          (my/bootstrap-ensure-seed-config-files))
         (t
          (message "Bootstrap: repo sync skipped (missing username or repo name)."))))
      ;; 7. git-crypt unlock (no-op if repo doesn't use it / already unlocked).
      ;; The key is in the same BW item as the rest of the GH identity,
      ;; under custom field "GitCryptKey".
      (my/git-crypt-step id)
      ;; 8. Starter data — only fills gaps; never overwrites pulled content
      (when (file-directory-p my/data-dir)
        (my/generate-starter-data my/data-dir id)
        (message "Bootstrap: starter files generated for any gaps under %s." my/data-dir)))
     (t
      ;; No GitHub identity — generate starter into a fresh my/data-dir
      (unless (file-directory-p my/data-dir)
        (make-directory my/data-dir t))
      (my/generate-starter-data my/data-dir '())
      (message "Bootstrap: GitHub identity unavailable; starter content generated locally at %s."
               my/data-dir))))

  ;; 8. API keys (visible progress emitted by my/api-keys-load itself)
  (my/api-keys-load)

  ;; 9. Lock BW
  (when my/bw-session (my/bw-lock))

  ;; 10. Drop the .bootstrap-completed marker so init.el skips the
  ;; URL-seed step on next launch. Delete this file manually to
  ;; force a re-seed of any missing files from the distro.
  (let ((marker (expand-file-name ".bootstrap-completed" my/config-dir)))
    (unless (file-exists-p marker)
      (with-temp-file marker
        (insert (format "Bootstrap completed at %s.

This file is a SENTINEL. Its existence tells init.el that
my/config-dir = %s is owned by you (the user) and to NOT re-seed
.org files from the public distro on subsequent launches.

Delete this file to force init.el to re-fetch any missing seed
files from %s on the next launch. Existing files in this directory
are NEVER overwritten — only missing ones are pulled.
"
                        (format-time-string "%Y-%m-%d %H:%M:%S %Z")
                        my/config-dir
                        my/distro-seed-url-base)))
      (message "Bootstrap: marker dropped at %s — next launch will skip seeding." marker)))

  (message "Bootstrap: done."))

;; Run it.
(my/bootstrap)
