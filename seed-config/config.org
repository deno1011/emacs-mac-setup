#+TITLE: Emacs Configuration
#+STARTUP: overview

Loaded by: [[file:~/.emacs.d/init.el][init.el]]

* Orchestration

This file is the entry point that =init.el= loads. It knows ONE thing:
that modules live in the =modules/= subdirectory next to this file,
discovered by listing the directory. It does not enumerate them.

** Convention

- =modules/NN-name.org= — module file. =NN= is a numeric prefix that
  decides load order; smaller loads first. Leave gaps between numbers
  (10, 20, 30) so future modules can insert without renumbering.
- =modules/NN-name.disabled.org= — disabled by rename. The file stays
  on disk but the discovery loop skips it.
- =local.org= (next to this file) — per-Mac overrides; loads LAST so it
  can override any module. Optional and untracked.

** Discovery

Module =10-bootstrap.org= must always run first (BW unlock, GitHub
identity, private-repo clone). On a fresh Mac the cloned repo can
introduce new modules into the working tree, so we re-scan the
directory after bootstrap finishes — that picks up renamed or newly
added modules without requiring a restart.

#+begin_src elisp :tangle yes
(defvar my/modules-dir
  (expand-file-name "modules/" my/config-dir)
  "Directory holding the module .org files discovered by config.org.")

(defun my/-module-files ()
  "Return the current sorted list of enabled module paths under
`my/modules-dir', skipping files whose basename matches
`.disabled.org'."
  (when (file-directory-p my/modules-dir)
    (sort (seq-remove
           (lambda (f) (string-match-p "\\.disabled\\.org\\'" f))
           (directory-files my/modules-dir t "\\.org\\'"))
          #'string<)))

(defun my/-load-module (path)
  "Load PATH, logging cleanly on error so one bad module does not abort
the whole config load."
  (condition-case err
      (org-babel-load-file path)
    (error
     (message "CONFIG LOAD ERROR (%s): %s"
              (file-name-nondirectory path) err))))

;; Pass 1: bootstrap only. Always present in the seeded modules/ and
;; runs first because it may clone the user's private repo on top of
;; the seeded config, replacing files we'd be about to load.
(let ((bootstrap (expand-file-name "modules/10-bootstrap.org" my/config-dir)))
  (when (file-exists-p bootstrap) (my/-load-module bootstrap)))

;; Pass 2: everything else. Re-scanned after bootstrap so renamed /
;; added modules from the just-cloned repo are seen immediately.
(dolist (f (my/-module-files))
  (unless (string-suffix-p "10-bootstrap.org" f)
    (my/-load-module f)))

;; Per-Mac override — loads LAST, can shadow anything above. Optional.
(let ((local (expand-file-name "local.org" my/config-dir)))
  (when (file-exists-p local) (my/-load-module local)))
#+end_src

* GTD Overlay

Loads the GTD configuration from =~/emacs/data/org/gtd-config.el=.
Runs AFTER all modules, so it overrides any org defaults set by
=50-org.org= or its successors.

#+begin_src elisp :tangle yes
(let ((gtd-config (expand-file-name "data/org/gtd-config.el" my/data-dir)))
  (when (file-exists-p gtd-config)
    (condition-case err
        (load gtd-config nil nil t)
      (error (message "GTD CONFIG LOAD ERROR: %s" err)))))
#+end_src
